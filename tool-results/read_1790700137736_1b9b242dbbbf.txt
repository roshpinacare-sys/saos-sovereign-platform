   220→// ─── v76 פנקס העניינים (THE ISSUES LEDGER) — the agents' own folder ───────
   221→// הוראת הבעלים: "תעלה את הסוגיה בתיקיה של הסוכנים" — the issues live in OUR
   222→// folder (issues/), and we read them from the file system on every boot and
   223→// patrol. The 401 of the chat can blind our mouth, never our eyes: the
   224→// ledger is read even when the gate is closed. Zero secrets — public issues
   225→// only; the format is the contract (see issues/README.md).
   226→
   227→interface LedgerIssue {
   228→  id: number;
   229→  title: string;
   230→  status: "open" | "closed";
   231→  priority: "high" | "normal" | "low";
   232→  ownerAction: boolean;
   233→  summary: string;
   234→  file: string;
   235→}
   236→
   237→function readIssues(): LedgerIssue[] {
   238→  try {
   239→    const files = fs
   240→      .readdirSync(ISSUES_DIR)
   241→      .filter((f) => f.endsWith(".md") && f !== "README.md")
   242→      .sort();
   243→    const out: LedgerIssue[] = [];
   244→    for (const f of files) {
   245→      try {
   246→        const raw = fs.readFileSync(path.join(ISSUES_DIR, f), "utf8");
   247→        const title = (/^#\s+(.+)$/m.exec(raw) ?? [])[1]?.trim();
   248→        const field = (k: string): string =>
   249→          (new RegExp(`^-\\s*${k}:\\s*(.*)$`, "m").exec(raw) ?? [])[1]?.trim() ?? "";
   250→        const id = Number(field("id"));
   251→        if (!title || !Number.isFinite(id) || id <= 0) continue; // torn file — skipped honestly
   252→        const priorityRaw = field("priority");
   253→        out.push({
   254→          id,
   255→          title,
   256→          status: field("status") === "closed" ? "closed" : "open",
   257→          priority: priorityRaw === "high" || priorityRaw === "low" ? priorityRaw : "normal",
   258→          ownerAction: /^yes$/i.test(field("owner-action")),
   259→          summary: field("summary").slice(0, 160),
   260→          file: f
   261→        });
   262→      } catch {
   263→        /* one torn file never breaks the patrol */
   264→      }
   265→    }
   266→    return out.sort((a, b) => a.id - b.id);
   267→  } catch {
   268→    return []; // no folder yet — an empty ledger is an honest ledger
   269→  }
   270→}

   272→const LEDGER = {
   273→  seen: new Map<number, string>() // id -> status (change detection across patrols)
   274→};
   275→
   276→function observeLedger(): {
   277→  issues: LedgerIssue[];
   278→  open: LedgerIssue[];
   279→  ownerAction: LedgerIssue[];
   280→  fresh: LedgerIssue[];
   281→  changed: LedgerIssue[];
   282→} {
   283→  const issues = readIssues();
   284→  const fresh: LedgerIssue[] = [];
   285→  const changed: LedgerIssue[] = [];
   286→  for (const i of issues) {
   287→    const prev = LEDGER.seen.get(i.id);
   288→    if (prev === undefined) fresh.push(i);
   289→    else if (prev !== i.status) changed.push(i);
   290→    LEDGER.seen.set(i.id, i.status);
   291→  }
   292→  return {
   293→    issues,
   294→    open: issues.filter((i) => i.status === "open"),
   295→    ownerAction: issues.filter((i) => i.status === "open" && i.ownerAction),
   296→    fresh,
   297→    changed
   298→  };
   299→}

   331→interface GateResponse {

   333→  ok: boolean;
   334→  error?: string;

   338→async function gateCall(a: AgentState, url: string, init: RequestInit | undefined, timeoutMs = 8_000): Promise<GateResponse> {
   339→  const headers: Record<string, string> = { "content-type": "application/json" };
   340→  if (a.identity) {
   341→    headers["x-agent-name"] = a.name;
   342→    headers["x-agent-key"] = a.identity.apiKey;
   343→  }
   344→  const res = await fetch(url, { ...init, headers: { ...headers, ...(init?.headers as object | undefined) }, signal: AbortSignal.timeout(timeoutMs) });
   345→  let body: { ok?: boolean; error?: string; messages?: FeedMessage[] } = {};
   346→  try {
   347→    body = (await res.json()) as typeof body;
   348→  } catch {
   349→    /* non-json — handled below */
   350→  }
   351→  return { status: res.status, ok: res.ok && body.ok !== false, error: body.error, messages: body.messages };
   352→}

   354→async function joinGate(a: AgentState): Promise<{ ok: boolean; apiKey?: string; error?: string }> {
   355→  try {
   356→    const res = await fetch(GATE, {
   357→      method: "POST",
   358→      headers: { "content-type": "application/json" },
   359→      body: JSON.stringify({ op: "join", name: a.name, role: a.role }),
   360→      signal: AbortSignal.timeout(8_000)
   361→    });
   362→    const j = (await res.json()) as { ok?: boolean; apiKey?: string; error?: string };
   363→    return { ok: !!j.ok, apiKey: j.apiKey, error: j.error };
   364→  } catch (e) {
   365→    return { ok: false, error: e instanceof Error ? e.message : "join unreachable" };
   366→  }
   367→}

   369→async function say(a: AgentState, content: string): Promise<boolean> {
   370→  if (!a.identity) return false;
   371→  const body = content.length > MAX_SAY ? `${content.slice(0, MAX_SAY - 3)}…` : content;
   372→  try {
   373→    const res = await gateCall(a, INBOX, { method: "POST", body: JSON.stringify({ content: body }) });
   374→    return res.ok;
   375→  } catch {
   376→    return false;
   377→  }
   378→}

   380→async function postOrder(a: AgentState, mission: string, difficulty: number, reward: number): Promise<boolean> {
   381→  if (!a.identity) return false;
   382→  try {
   383→    const res = await gateCall(a, INBOX, {
   384→      method: "POST",
   385→      body: JSON.stringify({ action: "order", mission, difficulty, reward })
   386→    });
   387→    if (res.ok) {
   388→      a.orders += 1;
   389→      a.lastMyOrderAt = Date.now();
   390→      journal(a.name, "order", `work order cut: "${mission.slice(0, 90)}" (difficulty ${difficulty}, reward ${reward}) — the soldier executes and is paid`);
   391→      return true;
   392→    }
   393→    journal(a.name, "order-refused", `order refused: ${res.error ?? res.status}`);
   394→    return false;
   395→  } catch (e) {
   396→    journal(a.name, "order-unreachable", e instanceof Error ? e.message.slice(0, 120) : "unknown");
   397→    return false;
   398→  }

   507→      }),

   522→    return { ok: false, latencyMs: Date.now() - t0 };

   838→interface Directive {
   839→  cmd: "work" | "analyze" | "patrol" | "report" | "hello" | "help";

   843→function parseDirective(a: AgentState, raw: string): Directive | null {
   844→  const content = raw.trim();
   845→  const lower = content.toLowerCase();
   846→  const at = lower.indexOf(`@${a.name}`);
   847→  if (at < 0) return null;
   848→  const rest = content
   849→    .slice(at + a.name.length + 1)
   850→    .replace(/^[\s:,.\u2013\-]+/, "")
   851→    .trim();
   852→  if (a.name === "gold-scout") {
   853→    if (/^(?:דווח|דוח|report)(?:\s|$)/i.test(rest) || rest === "דווח" || rest === "דוח") return { cmd: "work" };
   854→    const m = /^(?:נתח|analyze)\s*[:\-]?\s*(.*)$/i.exec(rest);
   855→    if (m) {
   856→      const topic = m[1].trim();
   857→      return topic ? { cmd: "analyze", arg: topic } : { cmd: "work" };
   858→    }
   859→    if (/^(?:שלום|היי|hello|hi)(?:\s|$)/i.test(rest)) return { cmd: "hello" };
   860→    return { cmd: "help" };
   861→  }
   862→  if (/^(?:סרוק|scan)(?:\s|$)/i.test(rest) || rest === "סרוק") return { cmd: "patrol" };
   863→  if (/^(?:דווח|דוח|report)(?:\s|$)/i.test(rest) || rest === "דווח" || rest === "דוח") return { cmd: "report" };
   864→  if (/^(?:שלום|היי|hello|hi)(?:\s|$)/i.test(rest)) return { cmd: "hello" };
   865→  return { cmd: "help" };
   866→}

   906→async function pollOnce(a: AgentState): Promise<void> {
   907→  if (!a.identity) {
   908→    const ok = await armIdentity(a, "first-boot");
   909→    if (!ok) return;
   910→  }
   911→  const t0 = Date.now();
   912→  let res: GateResponse;
   913→  try {
   914→    res = await gateCall(a, INBOX, { method: "GET" });
   915→  } catch (e) {
   916→    journal(a.name, "gate-down", `inbox unreachable: ${e instanceof Error ? e.message.slice(0, 100) : "unknown"} — honest wait, the keeper owns the web revive`);
   917→    return;
   918→  }
   919→  a.gateLatencyMs = Date.now() - t0;
   920→
   921→  if (res.status === 401) {
   922→    a.authFails += 1;
   923→    if ((res.error ?? "").includes("לא נמצא")) {
   924→      await armIdentity(a, "identity-wiped");
   925→    } else if (a.authFails % 20 === 1) {
   926→      journal(a.name, "auth-fail", `401 wrong key ×${a.authFails} (identity exists, hash differs — the owner must reissue)`);
   927→    }
   928→    return;
   929→  }
   930→  if (!res.ok) return;
   931→
   932→  a.polls += 1;
   933→  a.lastFeedAt = new Date().toISOString();
   934→  const feed = res.messages ?? [];
   935→  scanFeed(a, feed);
   936→  for (const m of feed) {
   937→    if (a.seen.includes(m.id)) continue;
   938→    a.seen.push(m.id);
   939→    if (a.seen.length > MAX_SEEN) a.seen = a.seen.slice(-Math.floor(MAX_SEEN / 2));
   940→    if (m.author === "owner" && m.kind === "say") {
   941→      const d = parseDirective(a, m.content);
   942→      if (d) {
   943→        a.directives += 1;
   944→        journal(a.name, "directive", `owner directive #${a.directives}: ${d.cmd}${d.arg ? ` — "${d.arg.slice(0, 80)}"` : ""}`);
   945→        await handleDirective(a, d);
   946→      }
   947→    }
   948→  }
   949→}

   992→// ─── v36 אכסניית-המושב-היחיד (SINGLE SEAT) ─────────────────────────────────
   993→// Root fix for the live double-boot incident: /start.sh re-ran over live
   994→// services and second instances lingered (zombies) or doubled work. The seat
   995→// is a PID lockfile in state/: a second instance exits CLEANLY when a live
   996→// owner actually serves (port answers — alive ≠ serving), takes over a stale
   997→// seat (dead owner), replaces a broken zombie (owner alive, port dead), and
   998→// always keeps its own PID (bun --hot re-eval). Fail-safe direction: at most
   999→// an honest "not starting", never a duplicate swarm host.
  1000→function acquireSeat(tag: string, probePort: number): void {
  1001→  const stateDir = path.join(import.meta.dir, "..", "state");
  1002→  const lockPath = path.join(stateDir, "seat.lock");
  1003→  try { fs.mkdirSync(stateDir, { recursive: true }); } catch { /* exists */ }
  1004→  const release = (): void => {
  1005→    try { if (fs.readFileSync(lockPath, "utf8").trim() === String(process.pid)) fs.rmSync(lockPath, { force: true }); } catch { /* not ours */ }
  1006→  };
  1007→  const GS = globalThis as unknown as { __seatExitBound?: boolean };
  1008→  if (!GS.__seatExitBound) { GS.__seatExitBound = true; process.on("exit", release); }
  1009→  const portAnswers = (): boolean => {
  1010→    try {
  1011→      const r = Bun.spawnSync(["bash", "-c", `exec 3<>/dev/tcp/127.0.0.1/${probePort} 2>/dev/null && exec 3<&- 3>&-`], { timeout: 1500 });
  1012→      return r.exitCode === 0;
  1013→    } catch { return false; }
  1014→  };
  1015→  for (let attempt = 1; attempt <= 3; attempt++) {
  1016→    try {
  1017→      const fd = fs.openSync(lockPath, "wx"); // atomic — EEXIST means the seat is taken
  1018→      fs.writeSync(fd, String(process.pid));
  1019→      fs.closeSync(fd);
  1020→      console.log(`[${tag}] seat acquired (pid ${process.pid}) — single-instance doctrine`);
  1021→      return;
  1022→    } catch (e) {
  1023→      const code = (e as { code?: string })?.code;
  1024→      if (code !== "EEXIST") throw e; // real IO error — crash honestly
  1025→      let ownerPid = 0;
  1026→      try { ownerPid = Number(fs.readFileSync(lockPath, "utf8").trim()); } catch { /* torn write */ }
  1027→      if (ownerPid === process.pid) return; // our own seat — bun --hot re-eval
  1028→      let ownerAlive = false;
  1029→      if (Number.isFinite(ownerPid) && ownerPid > 0) { try { process.kill(ownerPid, 0); ownerAlive = true; } catch { /* dead */ } }
  1030→      if (ownerAlive && portAnswers()) {
  1031→        console.log(`[${tag}] seat: pid ${ownerPid} already serves (port ${probePort} answers) — exiting cleanly, no duplicate`);
  1032→        process.exit(0);
  1033→      }
  1034→      console.log(`[${tag}] seat: ${ownerAlive ? `pid ${ownerPid} alive but NOT serving (zombie)` : `pid ${ownerPid} dead (stale)`} — taking over the seat`);
  1035→      try { fs.rmSync(lockPath, { force: true }); } catch { /* race — retry */ }
  1036→    }
  1037→  }
  1038→  console.log(`[${tag}] seat: contention unresolved — exiting cleanly, no duplicate`);
  1039→  process.exit(0);
  1040→}
  1041→acquireSeat("swarm-agents", PORT);
