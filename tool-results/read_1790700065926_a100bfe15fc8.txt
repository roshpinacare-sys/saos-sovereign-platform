   568→   568→interface TickerChip {
   569→   569→  id: string;
   570→   570→  name: string;
   571→   571→  value: string;

   576→   576→function dotTone(t: TickerChip["tone"]): string {
   577→   577→  switch (t) {
   578→   578→    case "gold":
   579→   579→      return "bg-amber-400";
   580→   580→    case "green":
   581→   581→      return "bg-emerald-400";
   582→   582→    case "amber":
   583→   583→      return "bg-amber-300";
   584→   584→    case "rose":
   585→   585→      return "bg-rose-400";
   586→   586→    default:
   587→   587→      return "bg-zinc-500";
   588→   588→  }
