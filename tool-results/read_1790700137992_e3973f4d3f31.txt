   122→function httpPostJson(url: string, body: string, timeoutMs = 15_000): Promise<unknown> {
   123→  const u = new URL(url);
   124→  return new Promise((resolve, reject) => {
   125→    const req = https.request(
   126→      {
   127→        hostname: u.hostname,
   128→        path: u.pathname || "/",
   129→        method: "POST",
   130→        headers: { "Content-Type": "application/json", "Content-Length": Buffer.byteLength(body) },
   131→        timeout: timeoutMs,
   132→        agent: false
   133→      },
   134→      (res) => {
   135→        const chunks: Buffer[] = [];
   136→        res.on("data", (c: Buffer) => chunks.push(c));
   137→        res.on("end", () => {
   138→          const data = Buffer.concat(chunks).toString("utf8");
   139→          if (res.statusCode !== 200) {
   140→            return reject(new Error(`HTTP ${res.statusCode} @ ${u.hostname}`));
   141→          }
   142→          try {
   143→            resolve(JSON.parse(data));
   144→          } catch {
   145→            reject(new Error(`bad json @ ${u.hostname}`));
   146→          }
   147→        });
   148→      }
   149→    );
   150→    req.on("timeout", () => req.destroy(new Error(`timeout @ ${u.hostname}`)));
   151→    req.on("error", reject);
   152→    req.write(body);
   153→    req.end();
   154→  });
   155→}

   158→  return httpPostJson(url, JSON.stringify({ jsonrpc: "2.0", id: 1, method, params })) as Promise<T>;
   159→}

   171→interface WeavePayload {
   172→  protocol: string | null;
   173→  h: number | null;
   174→  root: string | null;
   175→}
   176→
   177→function parseWeavePayload(raw: string | null | undefined): WeavePayload {
   178→  if (typeof raw !== "string" || raw.length === 0) return { protocol: null, h: null, root: null };
   179→  try {
   180→    const j = JSON.parse(raw) as { protocol?: unknown; h?: unknown; root?: unknown };
   181→    return {
   182→      protocol: typeof j.protocol === "string" ? j.protocol : null,
   183→      h: typeof j.h === "number" ? j.h : null,
   184→      root: typeof j.root === "string" ? j.root : null
   185→    };
   186→  } catch {
   187→    return { protocol: null, h: null, root: null };
   188→  }
   189→}

   191→/**
   192→ * הממו של TRON חוזר מ-gettransactionbyid כמחרוזת הקס (גם עם visible:true —
   193→ * נלמד חי בניסיון הראשון של החומה: '7b22616374696f6e…' = '{"action…').
   194→ * מנסים הקס קודם; אם הפענוח לא מזהה weave — מניחים שזה כבר utf8 גולמי.
   195→ */
   196→function decodeTronMemo(s: string): string | null {
   197→  if (/^[0-9a-fA-F]+$/.test(s) && s.length % 2 === 0 && s.length >= 8) {
   198→    try {
   199→      const decoded = Buffer.from(s, "hex").toString("utf8");
   200→      if (decoded.includes("saos-weave")) return decoded;
   201→    } catch {
   202→      // נופל להמשך — אולי זה utf8 גולמי
   203→    }
   204→  }
   205→  return s.includes("saos-weave") ? s : null;
   206→}

   238→async function verifyEvm(def: WallAnchorDef, rpcUrl: string): Promise<WallAnchorStatus> {
   239→  const at = new Date().toISOString();
   240→  const base: Omit<WallAnchorStatus, "ok"> = {
   241→    network: def.network,
   242→    heName: def.heName,
   243→    txid: def.txid,
   244→    explorer: def.explorer,
   245→    explorerName: def.explorerName,
   246→    firedIn: def.firedIn,
   247→    signatoryLabel: def.signatoryLabel,
   248→    reachable: false,
   249→    successMarker: null,
   250→    block: null,
   251→    blockMatches: null,
   252→    signatory: null,
   253→    signatoryMatches: null,
   254→    payloadProtocol: null,
   255→    sealHeight: null,
   256→    root: null,
   257→    endpoint: new URL(rpcUrl).hostname,
   258→    at
   259→  };
   260→  try {
   261→    const [receipt, tx] = await Promise.all([
   262→      evmRpc<{ status?: string; blockNumber?: string } | { result?: { status?: string; blockNumber?: string } }>(
   263→        rpcUrl,
   264→        "eth_getTransactionReceipt",
   265→        [def.txid]
   266→      ),
   267→      evmRpc<{ from?: string; input?: string } | { result?: { from?: string; input?: string } }>(
   268→        rpcUrl,
   269→        "eth_getTransactionByHash",
   270→        [def.txid]
   271→      )
   272→    ]);
   273→    const rc = ("result" in receipt && receipt.result ? receipt.result : receipt) as {
   274→      status?: string;
   275→      blockNumber?: string;
   276→    };
   277→    const txj = ("result" in tx && tx.result ? tx.result : tx) as { from?: string; input?: string };
   278→    if (!rc || (rc.status === undefined && rc.blockNumber === undefined)) {
   279→      return { ...base, ok: false, error: "receipt not found" };
   280→    }
   281→    const block = rc.blockNumber ? Number.parseInt(rc.blockNumber, 16) : null;
   282→    const from = typeof txj.from === "string" ? txj.from.toLowerCase() : null;
   283→    const payload = parseWeavePayload(
   284→      typeof txj.input === "string" && txj.input.startsWith("0x")
   285→        ? Buffer.from(txj.input.slice(2), "hex").toString("utf8")
   286→        : null
   287→    );
   288→    const successMarker = rc.status ?? null;
   289→    const signatoryMatches = from === null ? null : from === def.signatory;
   290→    const blockMatches = block === null || def.expectedBlock === null ? null : block === def.expectedBlock;
   291→    const ok =
   292→      successMarker === "0x1" &&
   293→      signatoryMatches === true &&
   294→      payload.protocol === "saos-weave/v1" &&
   295→      payload.root !== null &&
   296→      block !== null;
   297→    return {
   298→      ...base,
   299→      reachable: true,
   300→      successMarker,
   301→      block,
   302→      blockMatches,
   303→      signatory: from,
   304→      signatoryMatches,
   305→      payloadProtocol: payload.protocol,
   306→      sealHeight: payload.h,
   307→      root: payload.root,
   308→      ok
   309→    };
   310→  } catch (e) {
   311→    return { ...base, ok: false, error: e instanceof Error ? e.message : "evm verify failed" };
   312→  }
   313→}

   315→async function verifySolana(def: WallAnchorDef): Promise<WallAnchorStatus> {
   316→  const at = new Date().toISOString();
   317→  const base: Omit<WallAnchorStatus, "ok"> = {
   318→    network: def.network,
   319→    heName: def.heName,
   320→    txid: def.txid,
   321→    explorer: def.explorer,
   322→    explorerName: def.explorerName,
   323→    firedIn: def.firedIn,
   324→    signatoryLabel: def.signatoryLabel,
   325→    reachable: false,
   326→    successMarker: null,
   327→    block: null,
   328→    blockMatches: null,
   329→    signatory: null,
   330→    signatoryMatches: null,
   331→    payloadProtocol: null,
   332→    sealHeight: null,
   333→    root: null,
   334→    endpoint: new URL(SOL_RPC).hostname,
   335→    at
   336→  };
   337→  try {
   338→    const connection = new Connection(SOL_RPC, "confirmed");
   339→    const t = await connection.getTransaction(def.txid, { maxSupportedTransactionVersion: 0 });
   340→    if (!t) return { ...base, ok: false, error: "tx not found (history may be pruned on public RPC)" };
   341→    // התגובה מ-web3.js מגיעה מנותחת: accountKeys כאובייקטי PublicKey
   342→    // (לא מחרוזות) והוראות מפוענחות-חלקית (programId אובייקט + data
   343→    // Buffer) — מנרמלים הכל למחרוזות לפני ההשוואה (הבאג החי של הניסיון
   344→    // הראשון: PublicKey === string תמיד false).
   345→    const rawKeys =
   346→      (t.transaction.message as unknown as {
   347→        accountKeys?: Array<{ toBase58?: () => string } | string>;
   348→      }).accountKeys ?? [];
   349→    const keys: string[] = rawKeys.map((k) => (typeof k === "string" ? k : typeof k.toBase58 === "function" ? k.toBase58() : String(k)));
   350→    const feePayer = keys[0] ?? null;
   351→    const err = t.meta?.err ?? null;
   352→    const successMarker = err === null || err === undefined ? "err=null" : `err=${JSON.stringify(err).slice(0, 60)}`;
   353→    const block = typeof t.slot === "number" ? t.slot : null;
   354→    // ההוראה של תוכנית ה-Memo — המטען שלנו (data: Buffer מנותח או base58 גולמי)
   355→    const rawIxs =
   356→      (t.transaction.message as unknown as {
   357→        instructions?: Array<Record<string, unknown>>;
   358→      }).instructions ?? [];
   359→    let payloadRaw: string | null = null;
   360→    for (const ix of rawIxs) {
   361→      let programIdStr: string | null = null;
   362→      const pid = ix.programId as { toBase58?: () => string } | undefined;
   363→      if (pid && typeof pid.toBase58 === "function") {
   364→        programIdStr = pid.toBase58();
   365→      } else if (typeof ix.programIdIndex === "number") {
   366→        programIdStr = keys[ix.programIdIndex] ?? null;
   367→      }
   368→      if (programIdStr !== MEMO_PROGRAM) continue;
   369→      const data = ix.data as unknown;
   370→      if (Buffer.isBuffer(data)) {
   371→        payloadRaw = data.toString("utf8");
   372→      } else if (typeof data === "string") {
   373→        try {
   374→          payloadRaw = Buffer.from(bs58.decode(data)).toString("utf8");
   375→        } catch {
   376→          payloadRaw = null;
   377→        }
   378→      }
   379→    }
   380→    const payload = parseWeavePayload(payloadRaw);
   381→    const signatoryMatches = feePayer === null ? null : feePayer === def.signatory;
   382→    const blockMatches = block === null || def.expectedBlock === null ? null : block === def.expectedBlock;
   383→    const ok =
   384→      err === null &&
   385→      signatoryMatches === true &&
   386→      payload.protocol === "saos-weave/v1" &&
   387→      payload.root !== null &&
   388→      block !== null;
   389→    return {
   390→      ...base,
   391→      reachable: true,
   392→      successMarker,
   393→      block,
   394→      blockMatches,
   395→      signatory: feePayer,
   396→      signatoryMatches,
   397→      payloadProtocol: payload.protocol,
   398→      sealHeight: payload.h,
   399→      root: payload.root,
   400→      ok
   401→    };
   402→  } catch (e) {
   403→    return { ...base, ok: false, error: e instanceof Error ? e.message : "sol verify failed" };
   404→  }
   405→}

   431→      tronPost("/wallet/gettransactionbyid", { value: def.txid, visible: true }),
   432→      tronPost("/wallet/gettransactioninfobyid", { value: def.txid, visible: true })
   433→    ]);
   434→    const raw = txq as {
   435→      ret?: Array<{ contractRet?: string }>;
   436→      raw_data?: { data?: string };
   437→    };
   438→    const contractRet = raw.ret?.[0]?.contractRet ?? null;
   439→    if (contractRet === null && raw.raw_data === undefined) {
   440→      return { ...base, ok: false, error: "tx not found" };
   441→    }
   442→    const block = typeof (info as { blockNumber?: number }).blockNumber === "number"
   443→      ? (info as { blockNumber: number }).blockNumber
   444→      : null;
   445→    const owner =
   446→      (txq as { raw_data?: { contract?: Array<{ parameter?: { value?: { owner_address?: string } } }> } })
   447→        .raw_data?.contract?.[0]?.parameter?.value?.owner_address ?? null;
   448→    const payload = parseWeavePayload(
   449→      typeof raw.raw_data?.data === "string" ? decodeTronMemo(raw.raw_data.data) : null
   450→    );
   451→    const signatoryMatches = owner === null ? null : owner === def.signatory;
   452→    const blockMatches = block === null || def.expectedBlock === null ? null : block === def.expectedBlock;
   453→    const ok =
   454→      contractRet === "SUCCESS" &&
   455→      signatoryMatches === true &&
   456→      payload.protocol === "saos-weave/v1" &&
   457→      payload.root !== null &&
   458→      block !== null;
   459→    return {
   460→      ...base,
   461→      reachable: true,
   462→      successMarker: contractRet,
   463→      block,
   464→      blockMatches,
   465→      signatory: owner,
   466→      signatoryMatches,
   467→      payloadProtocol: payload.protocol,
   468→      sealHeight: payload.h,
   469→      root: payload.root,
   470→      ok
   471→    };
   472→  } catch (e) {
   473→    return { ...base, ok: false, error: e instanceof Error ? e.message : "tron verify failed" };
   474→  }
   475→}

   477→async function verifyXrpl(def: WallAnchorDef): Promise<WallAnchorStatus> {
   478→  const at = new Date().toISOString();

   491→    signatory: null,
   492→    signatoryMatches: null,
   493→    payloadProtocol: null,
   494→    sealHeight: null,
   495→    root: null,
   496→    endpoint: new URL(XRPL_RPC).hostname,
   497→    at
   498→  };

   506→        Memos?: Array<{ Memo?: { MemoData?: string } }>;

   517→    const account = typeof res.Account === "string" ? res.Account : null;

   519→    const payload = parseWeavePayload(
   520→      typeof memoHex === "string" ? Buffer.from(memoHex, "hex").toString("utf8") : null
   521→    );
   522→    const signatoryMatches = account === null ? null : account === def.signatory;
   523→    const blockMatches = block === null || def.expectedBlock === null ? null : block === def.expectedBlock;
   524→    const ok =
   525→      validated &&
   526→      engineResult === "tesSUCCESS" &&
   527→      signatoryMatches === true &&
   528→      payload.protocol === "saos-weave/v1" &&

   534→      successMarker: engineResult,
   535→      block,
   536→      blockMatches,
   537→      signatory: account,
   538→      signatoryMatches,
   539→      payloadProtocol: payload.protocol,
   540→      sealHeight: payload.h,
   541→      root: payload.root,
   542→      ok
   543→    };
   544→  } catch (e) {
   545→    return { ...base, ok: false, error: e instanceof Error ? e.message : "xrpl verify failed" };
   546→  }

   549→// ─── החומה — אימות כל חמשת העוגנים, עם מטמון 10 דק' ───────────────────────
   550→
   551→export interface WallStatus {
   552→  ok: boolean;
   553→  wall: WallAnchorStatus[];
   554→  verifiedCount: number;
   555→  reachableCount: number;
   556→  measuredAt: string;
   557→  tookMs: number;
   558→  cached: boolean;
   559→}
   560→
   561→const WALL_TTL_MS = 10 * 60_000;
   562→let wallCache: { at: number; value: WallStatus } | null = null;
   563→let wallInflight: Promise<WallStatus> | null = null;
   564→
   565→export async function verifyWall(opts: { force?: boolean } = {}): Promise<WallStatus> {
   566→  if (!opts.force && wallCache && Date.now() - wallCache.at < WALL_TTL_MS) {
   567→    return { ...wallCache.value, cached: true };
   568→  }
   569→  if (wallInflight) return wallInflight;
   570→  const started = Date.now();
   571→  wallInflight = (async () => {
   572→    const wall = await Promise.all(
   573→      WALL_ANCHORS.map((def) => {
   574→        switch (def.network) {
   575→          case "ETHEREUM":
   576→            return verifyEvm(def, ETH_RPC);
   577→          case "BSC":
   578→            return verifyEvm(def, BSC_RPC);
   579→          case "SOLANA":
   580→            return verifySolana(def);
   581→          case "TRON":
   582→            return verifyTron(def);
   583→          case "XRPL":
   584→            return verifyXrpl(def);
   585→        }
   586→      })
   587→    );
   588→    const value: WallStatus = {
   589→      ok: true,
   590→      wall,
   591→      verifiedCount: wall.filter((w) => w.ok).length,
   592→      reachableCount: wall.filter((w) => w.reachable).length,
   593→      measuredAt: new Date().toISOString(),
   594→      tookMs: Date.now() - started,
   595→      cached: false
   596→    };
   597→    wallCache = { at: Date.now(), value };
   598→    return value;
   599→  })();
   600→  try {
   601→    return await wallInflight;
   602→  } finally {
   603→    wallInflight = null;
   604→  }
   605→}

   607→// ─── שחזור הספר המקומי מאמת-השרשרת ────────────────────────────────────────
   608→
   609→export interface WallRebuildResult {
   610→  ok: boolean;
   611→  created: Array<{ network: string; txid: string; height: number | null; root: string | null; rowId: string }>;
   612→  existing: Array<{ network: string; txid: string }>;
   613→  skipped: Array<{ network: string; txid: string; reason: string }>;
   614→  receipt: string | null;
   615→  at: string;
   616→}
   617→
   618→/**
   619→ * בונה מחדש את שורות ה-AnchorTx המקומיות לחמשת עוגני החומה — אך ורק
   620→ * מתוך אימות חי מהשרשרת באותה נשימה (verifyWall force). שורה שכבר
   621→ * קיימת (לפי txid) לא נוגעת — הפעולה אידמפוטנטית. שורה שהשרשרת שלה
   622→ * לא אישרה כרגע נדלגת בכנות עם הסיבה.
   623→ */
   624→export async function rebuildWallLedger(): Promise<WallRebuildResult> {
   625→  const at = new Date().toISOString();
   626→  const status = await verifyWall({ force: true });
   627→  const created: WallRebuildResult["created"] = [];
   628→  const existing: WallRebuildResult["existing"] = [];
   629→  const skipped: WallRebuildResult["skipped"] = [];
   630→
   631→  for (const w of status.wall) {
   632→    const def = WALL_ANCHORS.find((d) => d.txid === w.txid)!;
   633→    const dupe = await db.anchorTx.findFirst({ where: { txid: def.txid } });
   634→    if (dupe) {
   635→      existing.push({ network: def.network, txid: def.txid });
   636→      continue;
   637→    }
   638→    if (!w.ok) {
   639→      skipped.push({
   640→        network: def.network,
   641→        txid: def.txid,
   642→        reason: w.error ?? "live verification did not pass — honest skip"
   643→      });
   644→      continue;
   645→    }
   646→    const row = await db.anchorTx.create({
   647→      data: {
   648→        sealId: null, // החותם המקורי חי ב-DB הקודם (טרום-איפוס) — כנות מלאה
   649→        network: def.network,
   650→        kind: "weave",
   651→        txid: def.txid,
   652→        status: "anchored",
   653→        gate: null,
   654→        meta: JSON.stringify({
   655→          live: true,
   656→          source: "wall-rebuild",
   657→          rebuiltFromChain: true,
   658→          note: "שורה נבנתה מחדש מאמת-השרשרת (v50) אחרי איפוס-הסביבה — העוגן עצמו מעולם לא נמחק, הוא חי במיינטנט",
   659→          height: w.sealHeight, // מספור החותמים של השרשרת המקומית הקודמת
   660→          root: w.root,
   661→          blockNumber: w.block,
   662→          blockMatches: w.blockMatches,
   663→          signatory: w.signatory,
   664→          signatoryLabel: def.signatoryLabel,
   665→          successMarker: w.successMarker,
   666→          payloadProtocol: w.payloadProtocol,
   667→          firedIn: def.firedIn,
   668→          explorer: def.explorer,
   669→          verifiedAt: w.at,
   670→          rebuiltAt: at
   671→        })
   672→      }
   673→    });
   674→    created.push({ network: def.network, txid: def.txid, height: w.sealHeight, root: w.root, rowId: row.id });
   675→  }
   676→
   677→  const receipt = await appendChainedReceipt(
   678→    "anchor-wall-rebuild",
   679→    {
   680→      created: created.map((c) => ({ network: c.network, txid: c.txid, height: c.height, root: c.root })),
   681→      existing: existing.length,
   682→      skipped: skipped.map((s) => ({ network: s.network, reason: s.reason })),
   683→      verifiedCount: status.verifiedCount
   684→    },
   685→    {}
   686→  ).catch(() => null);
   687→
   688→  if (created.length > 0 || skipped.length > 0) {
   689→    await logOp(
   690→      "anchor-wall-rebuild",
   691→      `created=${created.length} existing=${existing.length} skipped=${skipped.length} verified=${status.verifiedCount}/5`
   692→    ).catch(() => undefined);
   693→  }
   694→
   695→  return { ok: true, created, existing, skipped, receipt: receipt?.digest ?? null, at };
   696→}

   698→/**
   699→ * מפקד מקומי קטן — כמה שורות weave anchored יש כרגע בספר (להצגה
   700→ * לצד החומה: "מה השרשרת מחזיקה" מול "מה ה-DB יודע").

   702→export async function wallLedgerCount(): Promise<Record<string, number>> {
