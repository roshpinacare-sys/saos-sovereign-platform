     1→     1→/**
     2→     2→ * Saosmartwallet — SovereignVault state machine (M2, layer 1)
     3→     3→ * ==========================================================
     4→     4→ * The layer-1 owner of the persisted vault hierarchy: master vault, network
     5→     5→ * vaults, soldier fleet, purpose vaults, sealed keys, policy rule data and
     6→     6→ * backup bookkeeping — all inside ONE JSON state document (VaultStateDoc).
     7→     7→ *
     8→     8→ * Session model (fail closed):
     9→     9→ *   • Locked (fresh construction): everything is readable via getStatus()/
    10→    10→ *     getState() — those surfaces are SECRET-FREE by construction — but any
    11→    11→ *     mutation, key operation or export requires unlock(password).
    12→    12→ *   • unlock() re-derives the Argon2id master key into VOLATILE memory only
    13→    13→ *     and arms the idle auto-lock timer. lock() zero-fills the key bytes.
    14→    14→ *   • addKey({password}) performs an OPTIONAL step-up: the password is
    15→    15→ *     verified AGAIN against the stored verifier before a seal happens.
    16→    16→ *   • The derived key never persists, never travels in errors, and is wiped
    17→    17→ *     on lock — a stolen process snapshot after lock() has nothing.
    18→    18→ *
    19→    19→ * Persistence: storage-agnostic (VaultStorage seam). The genesis hash binds
    20→    20→ * all saves of one vault instance (sha256 of the stable vaultInstanceId) so
    21→    21→ * the audit layer can anchor its chain to a specific state lineage.
    22→    22→ */

   222→   222→export class SovereignVault {
   223→   223→  private readonly storage: VaultStorage;
   224→   224→  private readonly audit: AuditStore | null;
   225→   225→  private readonly nowFn: () => number;
   226→   226→
   227→   227→  /** Session master key — volatile, zero-filled on lock(). */
   228→   228→  private derivedKey: Uint8Array | null = null;
   229→   229→  private unlockedAt: number | null = null;
   230→   230→  private lastActivityAt: number | null = null;
   231→   231→  private autoLocksAt: number | null = null;
   232→   232→  /** Cached state document (persisted copy lives in the VaultStorage seam). */
   233→   233→  private state: VaultStateDoc | null = null;
   234→   234→  private genesisHashValue: string | null = null;
   235→   235→
   236→   236→  /**
   237→   237→   * @param storage the VaultStorage seam (memory store for tools/tests,
   238→   238→   *   Prisma-backed store in the app layer).
   239→   239→   * @param audit optional AuditStore — read-only here for status telemetry
   240→   240→   *   (head hash + event count); chain writing belongs to the audit layer.
   241→   241→   */
   242→   242→  constructor(storage: VaultStorage, audit?: AuditStore, opts?: SovereignVaultOptions) {
   243→   243→    this.storage = storage;
   244→   244→    this.audit = audit ?? null;
   245→   245→    this.nowFn = opts?.now ?? (() => Date.now());
   246→   246→  }
   247→   247→
   248→   248→  // -------------------------------------------------------------------------
   249→   249→  // Lifecycle
   250→   250→  // -------------------------------------------------------------------------
   251→   251→
   252→   252→  /**
   253→   253→   * Loads (or reloads) the state document from the storage seam. Safe to call
   254→   254→   * repeatedly; a missing record leaves the vault in the uninitialized state.
   255→   255→   *
   256→   256→   * @throws VaultError("STORAGE_FAILURE") on backend errors,
   257→   257→   *   VaultError("CORRUPT_STATE") when the stored document is not valid
   258→   258→   *   docVersion-1 JSON (fail closed — never run against a half-parsed state).
   259→   259→   */
   260→   260→  async load(): Promise<void> {
   261→   261→    let record: VaultStateRecord | null = null;
   262→   262→    try {
   263→   263→      record = await this.storage.load();
   264→   264→    } catch {
   265→   265→      throw new VaultError("STORAGE_FAILURE", "failed to read the vault state record");
   266→   266→    }
   267→   267→    if (record === null) {
   268→   268→      this.state = null;
   269→   269→      this.genesisHashValue = null;
   270→   270→      return;
   271→   271→    }
   272→   272→    let parsed: unknown;
   273→   273→    try {
   274→   274→      parsed = JSON.parse(record.stateJson);
   275→   275→    } catch {
   276→   276→      throw new VaultError("CORRUPT_STATE", "stored vault state is not valid JSON");
   277→   277→    }
   278→   278→    if (!isVaultStateDoc(parsed)) {
   279→   279→      throw new VaultError("CORRUPT_STATE", "stored vault state does not match the docVersion 1 schema");
   280→   280→    }
   281→   281→    this.state = parsed;
   282→   282→    this.genesisHashValue = null; // re-derived lazily from the loaded instance id
   283→   283→  }
   284→   284→
   285→   285→  /**
   286→   286→   * First-boot ceremony: installs the Argon2id password verifier and the full
   287→   287→   * M2 ecosystem — master vault (3-of-5 guardians, multi-sig), the five
   288→   288→   * network vaults (steem/hive/tron/blurt/saos-net), the soldier fleet
   289→   289→   * (13 by default, 5 traders), and the four purpose vaults
   290→   290→   * (trading/bridging/governance/treasury).
   291→   291→   *
   292→   292→   * The password itself is consumed here and never stored — only its verifier.
   293→   293→   *
   294→   294→   * @throws VaultError("VAULT_EXISTS") when already initialized,
   295→   295→   *   VaultError("VALIDATION") for malformed option amounts.
   296→   296→   */
   297→   297→  async initialize(password: string, opts?: InitializeOptions): Promise<VaultStatus> {
   298→   298→    await this.ensureLoaded();
   299→   299→    if (this.state === null) {
   300→   300→      // First boot — no persisted document yet: start from a fresh skeleton
   301→   301→      // with a stable random instance id (binds audit genesis + ZK contexts).
   302→   302→      this.state = emptyState(crypto.randomUUID());
   303→   303→    }
   304→   304→    const state = this.state;
   305→   305→    if (state.initialized) {
   306→   306→      throw new VaultError("VAULT_EXISTS", "vault is already initialized");
   307→   307→    }
   308→   308→    if (opts?.steemDailyLimitSatoshis !== undefined) {
   309→   309→      try {
   310→   310→        parseSats(opts.steemDailyLimitSatoshis);
   311→   311→      } catch {
   312→   312→        throw new VaultError("VALIDATION", "steemDailyLimitSatoshis is not a valid satoshi string");
   313→   313→      }
   314→   314→    }
   315→   315→    const soldierCount = opts?.soldierCount ?? 13;
   316→   316→    const traderSoldiers = opts?.traderSoldiers ?? 5;
   317→   317→    if (!Number.isInteger(soldierCount) || soldierCount < 1 || soldierCount > 999) {
   318→   318→      throw new VaultError("VALIDATION", "soldierCount must be an integer in [1, 999]");
   319→   319→    }
   320→   320→    if (!Number.isInteger(traderSoldiers) || traderSoldiers < 0 || traderSoldiers > soldierCount) {
   321→   321→      throw new VaultError("VALIDATION", "traderSoldiers must be an integer in [0, soldierCount]");
   322→   322→    }
   323→   323→
   324→   324→    const now = this.nowFn();
   325→   325→
   326→   326→    // Master vault — root of trust, ed25519, 3-of-5 guardian multi-sig.
   327→   327→    state.master = {
   328→   328→      id: "master",
   329→   329→      kind: "master",
   330→   330→      label: "Master vault",
   331→   331→      createdAt: now,
   332→   332→      approvers: ["master"],
   333→   333→      keyType: "ed25519",
   334→   334→      requiresMultiSig: true,
   335→   335→      threshold: 3,
   336→   336→      guardianCount: 5,
   337→   337→    };
   338→   338→
   339→   339→    // Network vaults — blurt ships offline; saos-net carries the sovereign flags.
   340→   340→    const networks: Record<string, NetworkVaultConfig> = {};
   341→   341→    const networkConfigs = [
   342→   342→      this.buildNetworkConfig(
   343→   343→        { network: "steem", maxAmountPerDay: { STEEM: opts?.steemDailyLimitSatoshis ?? "10000000" } },
   344→   344→        now,
   345→   345→      ),
   346→   346→      this.buildNetworkConfig({ network: "hive" }, now),
   347→   347→      this.buildNetworkConfig({ network: "tron" }, now),
   348→   348→      this.buildNetworkConfig({ network: "blurt", status: "offline" }, now),
   349→   349→      this.buildNetworkConfig(
   350→   350→        { network: "saos-net", special: { soldierManagement: true, oracleFeeds: true } },
   351→   351→        now,
   352→   352→      ),
   353→   353→      // M3: EVM + Solana vaults — born offline until a key is imported,
   354→   354→      // exactly like blurt (honest until funded + keyed).
   355→   355→      this.buildNetworkConfig({ network: "ethereum", status: "offline" }, now),
   356→   356→      this.buildNetworkConfig({ network: "solana", status: "offline" }, now),
   357→   357→    ];
   358→   358→    for (const cfg of networkConfigs) networks[cfg.id] = cfg;
   359→   359→
   360→   360→    // Soldier fleet — ids and names zero-padded; the first N are traders.
   361→   361→    const soldiers: Record<string, SoldierVaultConfig> = {};
   362→   362→    for (let i = 1; i <= soldierCount; i++) {
   363→   363→      const seq = String(i).padStart(2, "0");
   364→   364→      soldiers[`vault-soldier-${seq}`] = {
   365→   365→        id: `vault-soldier-${seq}`,
   366→   366→        kind: "soldier",
   367→   367→        label: `saos-soldier-${seq}`,
   368→   368→        createdAt: now,
   369→   369→        approvers: ["master"],
   370→   370→        soldierName: `saos-soldier-${seq}`,
   371→   371→        permissions: {
   372→   372→          canTrade: i <= traderSoldiers,
   373→   373→          canVote: true,
   374→   374→          canComment: true,
   375→   375→          canTransfer: false,
   376→   376→          maxTransferAmount: "0",
   377→   377→        },
   378→   378→        linkedVaults: [...DEFAULT_SOLDIER_LINKS],
   379→   379→        anomalyDetection: true,
   380→   380→        lastActivity: 0,
   381→   381→      };
   382→   382→    }
   383→   383→
   384→   384→    // Purpose vaults — treasury is air-gapped (approval via policy rules).
   385→   385→    const purposes: Record<string, PurposeVaultConfig> = {};
   386→   386→    for (const purpose of ["trading", "bridging", "governance", "treasury"] as const) {
   387→   387→      const d = PURPOSE_DEFAULTS[purpose];
   388→   388→      purposes[`purpose-${purpose}`] = {
   389→   389→        id: `purpose-${purpose}`,
   390→   390→        kind: "purpose",
   391→   391→        label: `${purpose} purpose vault`,
   392→   392→        createdAt: now,
   393→   393→        approvers: ["master"],
   394→   394→        purpose,
   395→   395→        allowedOperations: [...d.allowedOperations],
   396→   396→        networkScope: [...d.networkScope],
   397→   397→        storage: d.storage,
   398→   398→      };
   399→   399→    }
   400→   400→
   401→   401→    state.networks = networks;
   402→   402→    state.soldiers = soldiers;
   403→   403→    state.purposes = purposes;
   404→   404→    state.verifier = await deriveVerifier(password);
   405→   405→    // Layer-2 enrollment: the curve commitment binds the SAME password into an
   406→   406→    // independent verifier, anchored to this vault instance's ZK context.
   407→   407→    state.zkCommitment = deriveCommitment(password, state.verifier.salt, zkUnlockContext(state.vaultInstanceId));
   408→   408→    state.initialized = true;
   409→   409→
   410→   410→    await this.persist();
   411→   411→    return this.getStatus();
   412→   412→  }
   413→   413→
   414→   414→  /**
   415→   415→   * Verifies the master password against the stored Argon2id verifier and, on
   416→   416→   * success, holds the derived 32-byte master key in VOLATILE memory until
   417→   417→   * lock() wipes it. Also arms the idle auto-lock timer.
   418→   418→   *
   419→   419→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("WRONG_PASSWORD") — the
   420→   420→   *   error reveals only that verification failed, never why or what was stored.
   421→   421→   */
   422→   422→  async unlock(password: string): Promise<VaultStatus> {
   423→   423→    await this.ensureLoaded();
   424→   424→    const state = this.requireState();
   425→   425→    if (!state.initialized || state.verifier === null) {
   426→   426→      throw new VaultError("VAULT_NOT_INITIALIZED", "vault has no password verifier — initialize() first");
   427→   427→    }
   428→   428→    // Layer 2 first — cheap curve math fails fast and detects commitment tamper.
   429→   429→    await this.assertZKPasswordKnowledge(password, state);
   430→   430→    const ok = await verifyPassword(password, state.verifier);
   431→   431→    if (!ok) {
   432→   432→      throw new VaultError("WRONG_PASSWORD", "password verification failed");
   433→   433→    }
   434→   434→    this.derivedKey = await deriveMasterKey(password, b64ToBytes(state.verifier.salt), state.verifier.params);
   435→   435→    const t = this.nowFn();
   436→   436→    this.unlockedAt = t;
   437→   437→    this.lastActivityAt = t;
   438→   438→    this.autoLocksAt = t + state.autoLockSeconds * 1000;
   439→   439→    return this.getStatus();
   440→   440→  }
   441→   441→
   442→   442→  /**
   443→   443→   * Locks the session: zero-fills the derived key bytes, clears all volatile
   444→   444→   * session timestamps. Idempotent and safe to call at any time.
   445→   445→   */
   446→   446→  lock(): void {
   447→   447→    if (this.derivedKey !== null) {
   448→   448→      this.derivedKey.fill(0);
   449→   449→      this.derivedKey = null;
   450→   450→    }
   451→   451→    this.unlockedAt = null;
   452→   452→    this.autoLocksAt = null;
   453→   453→  }
   454→   454→
   455→   455→  /**
   456→   456→   * Auto-lock check — call from a timer/poll. Locks and returns true when the
   457→   457→   * idle window has elapsed; otherwise a no-op returning false.
   458→   458→   */
   459→   459→  maybeAutoLock(now: number = Date.now()): boolean {
   460→   460→    if (this.derivedKey !== null && this.autoLocksAt !== null && now > this.autoLocksAt) {
   461→   461→      this.lock();
   462→   462→      return true;
   463→   463→    }
   464→   464→    return false;
   465→   465→  }
   466→   466→
   467→   467→  /**
   468→   468→   * Sliding-window activity bump: refreshes lastActivityAt and re-arms the
   469→   469→   * auto-lock deadline. Called automatically by ensureUnlocked().
   470→   470→   */
   471→   471→  touch(): void {
   472→   472→    const t = this.nowFn();
   473→   473→    this.lastActivityAt = t;
   474→   474→    this.autoLocksAt = t + this.autoLockSeconds() * 1000;
   475→   475→  }

   478→   478→   * Session gate for every privileged operation: throws VaultError("VAULT_LOCKED")

   485→   485→    this.touch();
   486→   486→  }

   650→   650→  /**
   651→   651→   * Seals a WIF into the vault under the session master key (see VaultKeys).
   652→   652→   *
   653→   653→   * Step-up: when `input.password` is provided it is verified AGAIN against
   654→   654→   * the stored verifier — a failed step-up aborts the seal with
   655→   655→   * VaultError("WRONG_PASSWORD").
   656→   656→   *
   657→   657→   * Returns metadata WITHOUT the sealed payload (the caller never needs the
   658→   658→   * ciphertext; reveal goes through revealKeyFor).
   659→   659→   *
   660→   660→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("VAULT_LOCKED"),
   661→   661→   *   ("WRONG_PASSWORD"), ("VALIDATION") for bad WIFs/roles.
   662→   662→   */
   663→   663→  async addKey(input: VaultAddKeyInput): Promise<Omit<EncryptedKey, "payload">> {
   664→   664→    await this.ensureLoaded();
   665→   665→    const state = this.requireState();
   666→   666→    this.ensureUnlocked();
   667→   667→    if (input.password !== undefined) {
   668→   668→      if (state.verifier === null) {
   669→   669→        throw new VaultError("VAULT_NOT_INITIALIZED", "vault has no password verifier");
   670→   670→      }
   671→   671→      await this.assertZKPasswordKnowledge(input.password, state);
   672→   672→      const ok = await verifyPassword(input.password, state.verifier);
   673→   673→      if (!ok) {
   674→   674→        throw new VaultError("WRONG_PASSWORD", "step-up password verification failed");
   675→   675→      }
   676→   676→    }
   677→   677→    const { record, state: nextState } = VaultKeys.addKey(state, this.requireDerivedKey(), input);
   678→   678→    this.state = nextState;
   679→   679→    await this.persist();
   680→   680→    return {
   681→   681→      keyId: record.keyId,
   682→   682→      vaultId: record.vaultId,
   683→   683→      network: record.network,
   684→   684→      role: record.role,
   685→   685→      account: record.account,
   686→   686→      label: record.label,
   687→   687→      algorithm: record.algorithm,
   688→   688→      publicKey: record.publicKey,
   689→   689→      storage: record.storage,
   690→   690→      createdAt: record.createdAt,
   691→   691→      lastUsedAt: record.lastUsedAt,
   692→   692→      status: record.status,
   693→   693→    };
   694→   694→  }
   695→   695→
   696→   696→  /**
   697→   697→   * The ONLY sanctioned plaintext exit for a sealed key: unlock + policy +
   698→   698→   * revocation checks happen HERE, before VaultKeys.revealKey ever runs.
   699→   699→   * Refreshes session activity (idle-timer discipline for signing flows).
   700→   700→   *
   701→   701→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("VAULT_LOCKED"),
   702→   702→   *   ("KEY_NOT_FOUND"), ("KEY_REVOKED"), ("DECRYPTION_FAILED").
   703→   703→   */
   704→   704→  async revealKeyFor(keyId: string): Promise<RevealResult> {
   705→   705→    await this.ensureLoaded();
   706→   706→    const state = this.requireState();
   707→   707→    this.ensureUnlocked();
   708→   708→    const record = state.keys[keyId];
   709→   709→    if (!record) {
   710→   710→      throw new VaultError("KEY_NOT_FOUND", "no such key in the vault");
   711→   711→    }
   712→   712→    if (record.status === "revoked") {
   713→   713→      throw new VaultError("KEY_REVOKED", "key has been revoked");
   714→   714→    }
   715→   715→    const privateKey = VaultKeys.revealKey(record, this.requireDerivedKey());
   716→   716→    return { privateKey, record };
   717→   717→  }
   718→   718→
   719→   719→  /**
   720→   720→   * Records a signing use (lastUsedAt) for audit + dead-man telemetry.
   721→   721→   * Metadata-only: deliberately does NOT require an unlocked session.
   722→   722→   *
   723→   723→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("KEY_NOT_FOUND").
   724→   724→   */
   725→   725→  async markKeyUsed(keyId: string): Promise<void> {
   726→   726→    await this.ensureLoaded();
   727→   727→    const state = this.requireState();
   728→   728→    const record = state.keys[keyId];
   729→   729→    if (!record) {
   730→   730→      throw new VaultError("KEY_NOT_FOUND", "no such key in the vault");
   731→   731→    }
   732→   732→    state.keys[keyId] = { ...record, lastUsedAt: this.nowFn() };
   733→   733→    await this.persist();
   734→   734→  }

   736→   736→  /**
   737→   737→   * Step-up authentication gate (used by the signing orchestrator and the
   738→   738→   * backup manager): re-verifies a password against the stored Argon2id
   739→   739→   * verifier — the SAME pattern addKey uses for its optional step-up.
   740→   740→   *
   741→   741→   * @throws VaultError("VAULT_NOT_INITIALIZED") when no verifier exists,
   742→   742→   *   VaultError("WRONG_PASSWORD") when verification fails. The error
   743→   743→   *   reveals only that verification failed — never what was stored.
   744→   744→   */
   745→   745→  async verifyStepUp(password: string): Promise<void> {
   746→   746→    await this.ensureLoaded();
   747→   747→    const state = this.requireState();
   748→   748→    if (!state.initialized || state.verifier === null) {
   749→   749→      throw new VaultError("VAULT_NOT_INITIALIZED", "vault has no password verifier — initialize() first");
   750→   750→    }
   751→   751→    await this.assertZKPasswordKnowledge(password, state);
   752→   752→    const ok = await verifyPassword(password, state.verifier);
   753→   753→    if (!ok) {
   754→   754→      throw new VaultError("WRONG_PASSWORD", "step-up password verification failed");
   755→   755→    }
   756→   756→  }
   757→   757→
   758→   758→  /**
   759→   759→   * Zero-knowledge password-knowledge check (M2 layer 2 — defense in depth).
   760→   760→   * Generates a Fiat–Shamir Schnorr transcript for `password` and verifies it
   761→   761→   * against the curve commitment installed at initialize(). This is an
   762→   762→   * INDEPENDENT second verifier: unlock/step-up now require BOTH the ZK
   763→   763→   * commitment check AND the Argon2id verifier to agree on the same password,
   764→   764→   * so corruption or mutation of either stored verifier alone fails closed
   765→   765→   * with WRONG_PASSWORD (no oracle about which check rejected).
   766→   766→   *
   767→   767→   * @throws VaultError("VAULT_NOT_INITIALIZED") when no verifier exists,
   768→   768→   *   VaultError("CORRUPT_STATE") when an initialized vault lost its
   769→   769→   *   commitment (restore from backup or re-initialize),
   770→   770→   *   VaultError("WRONG_PASSWORD") when the transcript does not verify.
   771→   771→   */
   772→   772→  private async assertZKPasswordKnowledge(password: string, state: VaultStateDoc): Promise<void> {
   773→   773→    if (!state.initialized || state.verifier === null) {
   774→   774→      throw new VaultError("VAULT_NOT_INITIALIZED", "vault has no password verifier — initialize() first");
   775→   775→    }
   776→   776→    if (state.zkCommitment === null) {
   777→   777→      throw new VaultError("CORRUPT_STATE", "vault state is missing its ZK commitment — restore from backup or re-initialize");
   778→   778→    }
   779→   779→    const proof = await generateZKProof(password, state.verifier.salt, zkUnlockContext(state.vaultInstanceId));
   780→   780→    const ok = await verifyZKProof(proof, state.zkCommitment);
   781→   781→    if (!ok) {
   782→   782→      throw new VaultError("WRONG_PASSWORD", "password knowledge check failed");
   783→   783→    }
   784→   784→  }
   785→   785→
   786→   786→  /**
   787→   787→   * Key-selection lookup for the signing orchestrator: the keyId of the first
   788→   788→   * ACTIVE key matching (vaultId, network, role), oldest creation first.
   789→   789→   * Metadata-only read — works while LOCKED (the payload stays sealed; the
   790→   790→   * caller still needs revealKeyFor + an unlocked session to use the key).
   791→   791→   *
   792→   792→   * @returns the keyId, or null when no active key matches.
   793→   793→   */
   794→   794→  selectKeyId(input: { vaultId: string; network: NetworkId; role: KeyRole | null }): string | null {
   795→   795→    const state = this.state;
   796→   796→    if (state === null) return null;
   797→   797→    let best: EncryptedKey | null = null;
   798→   798→    for (const record of Object.values(state.keys)) {
   799→   799→      if (record.status !== "active") continue;
   800→   800→      if (record.vaultId !== input.vaultId || record.network !== input.network) continue;
   801→   801→      if (record.role !== input.role) continue;
   802→   802→      if (best === null || record.createdAt < best.createdAt) best = record;
   803→   803→    }
   804→   804→    return best?.keyId ?? null;
   805→   805→  }

   809→   809→   * VaultKeys.revokeKey. Records persist (revocation history is audit
   810→   810→   * evidence); idempotent on already-revoked keys. Requires an unlocked
   811→   811→   * session — the key material stays sealed.
   812→   812→   *
   813→   813→   * @returns the revoked record's metadata WITHOUT the sealed payload.
   814→   814→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("VAULT_LOCKED"),
   815→   815→   *   ("KEY_NOT_FOUND") for an unknown keyId.

   823→   823→    await this.persist();

   825→   825→    return metadata;
   826→   826→  }

   858→   858→  /**
   859→   859→   * Re-seals the ENTIRE state document JSON under a freshly Argon2id-derived
   860→   860→   * key (fresh 16-byte salt, default params). The payload's `salt` field
   861→   861→   * carries the REAL derivation salt, so a restorer needs only the password:
   862→   862→   *   key = deriveMasterKey(password, b64ToBytes(payload.salt))
   863→   863→   *   state = openJson(payload, key)
   864→   864→   *
   865→   865→   * This is the input format BackupManager (layer 5) wraps into a
   866→   866→   * BackupPackage with Shamir shards of the wrapping key.
   867→   867→   *
   868→   868→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("VAULT_LOCKED"),
   869→   869→   *   ("VALIDATION") for an empty password.
   870→   870→   */
   871→   871→  async exportState(password: string): Promise<EncryptedPayload> {
   872→   872→    await this.ensureLoaded();
   873→   873→    const state = this.requireState();
   874→   874→    this.ensureUnlocked();
   875→   875→    if (password.length === 0) {
   876→   876→      throw new VaultError("VALIDATION", "backup password must not be empty");
   877→   877→    }
   878→   878→    const salt = randomBytes(16);
   879→   879→    const key = await deriveMasterKey(password, salt, DEFAULT_KDF_PARAMS);
   880→   880→    try {
   881→   881→      return seal(utf8(JSON.stringify(state)), key, { keyId: "backup-v1", salt });
   882→   882→    } finally {
   883→   883→      key.fill(0);
   884→   884→    }
   885→   885→  }

   887→   887→  /**
   888→   888→   * Persists the social-recovery CONFIGURATION (contacts + threshold) into
   889→   889→   * `state.backup.socialRecovery`. Metadata only — the Shamir shares and the
   890→   890→   * R-sealed state payload NEVER persist here; the BackupManager returns
   891→   891→   * them to the caller for out-of-band storage.
   892→   892→   *
   893→   893→   * @throws VaultError("VAULT_NOT_INITIALIZED"), ("VAULT_LOCKED").

   930→   930→  async getStatus(): Promise<VaultStatus> {

   932→   932→    const state = this.state;
   933→   933→    if (state === null) {
   934→   934→      return {
   935→   935→        initialized: false,
   936→   936→        locked: true,
   937→   937→        autoLockSeconds: DEFAULT_AUTO_LOCK_SECONDS,
   938→   938→        autoLocksAt: null,
   939→   939→        createdAt: null,
   940→   940→        securityScore: 0,
   941→   941→        securityFindings: ["vault is not initialized — run initialize() to set a master password verifier"],

   951→   951→    }

   954→   954→    const activeKeys = keys.filter((k) => k.status === "active");

   961→   961→    const { score, findings } = this.computeSecurityScore(state);
   962→   962→

   991→   991→  /**
   992→   992→   * Honest security posture scoring (0–100) from REAL configuration only —
   993→   993→   * never defaults that make an uninitialized vault look good. Findings are
   994→   994→   * short, actionable, and contain no secrets.
   995→   995→   *
   996→   996→   * Budget: 20 base + 15 verifier + 10 keys + 10 steem role coverage +
   997→   997→   * 10 cold/air-gapped storage + 10 backup + 10 social recovery +
   998→   998→   * 10 dead-man switch + 5 tight auto-lock = 100.
   999→   999→   */
  1000→  1000→  computeSecurityScore(state: VaultStateDoc): { score: number; findings: string[] } {
  1001→  1001→    if (!state.initialized) {
  1002→  1002→      return {
  1003→  1003→        score: 0,
  1004→  1004→        findings: ["vault is not initialized — run initialize() to set a master password verifier"],
  1005→  1005→      };
  1006→  1006→    }
  1007→  1007→    const findings: string[] = [];
  1008→  1008→    let score = 20;
  1009→  1009→
  1010→  1010→    const keys = Object.values(state.keys);
  1011→  1011→    if (state.verifier !== null) score += 15;
  1012→  1012→    else findings.push("no master-password verifier present — the vault cannot verify unlock attempts");
  1013→  1013→
  1014→  1014→    if (keys.length > 0) score += 10;
  1015→  1015→    else findings.push("no keys sealed yet — add at least one network key");
  1016→  1016→
  1017→  1017→    const steemCfg = Object.values(state.networks).find((n) => n.network === "steem");
  1018→  1018→    const roles = new Set(steemCfg?.roles ?? []);
  1019→  1019→    if (STEEM_ROLES.every((r) => roles.has(r))) score += 10;
  1020→  1020→    else findings.push("steem vault lacks full role coverage (owner/active/posting/memo)");
  1021→  1021→
  1022→  1022→    if (keys.some((k) => k.storage !== "hot")) score += 10;
  1023→  1023→    else findings.push("all keys are hot-stored — seal at least one cold or air-gapped key");
  1024→  1024→
  1025→  1025→    if (state.backup.lastBackupAt !== null) score += 10;
  1026→  1026→    else findings.push("no backup created yet — export the state doc via exportState()");
  1027→  1027→
  1028→  1028→    const recovery = state.backup.socialRecovery;
  1029→  1029→    if (recovery !== null && recovery.threshold > 0 && recovery.contacts.length >= recovery.threshold) score += 10;
  1030→  1030→    else findings.push("social recovery is not configured — register recovery contacts up to a threshold");
  1031→  1031→
  1032→  1032→    if (state.backup.deadManSwitch?.enabled) score += 10;
  1033→  1033→    else findings.push("dead-man switch is disabled — configure beneficiaries and an inactivity window");
  1034→  1034→
  1035→  1035→    if (state.autoLockSeconds <= 300) score += 5;
  1036→  1036→    else findings.push(`auto-lock of ${state.autoLockSeconds}s exceeds the 300s budget`);
  1037→  1037→
  1038→  1038→    return { score: Math.max(0, Math.min(100, score)), findings };
  1039→  1039→  }

  1111→  1111→/** Structural guard for the persisted state document (docVersion 1). */
  1112→  1112→function isVaultStateDoc(v: unknown): v is VaultStateDoc {
  1113→  1113→  if (v === null || typeof v !== "object") return false;
  1114→  1114→  const o = v as Record<string, unknown>;
  1115→  1115→  return (
  1116→  1116→    o.docVersion === 1 &&
  1117→  1117→    typeof o.vaultInstanceId === "string" &&
  1118→  1118→    typeof o.initialized === "boolean" &&
  1119→  1119→    typeof o.createdAt === "number" &&
  1120→  1120→    typeof o.updatedAt === "number" &&
  1121→  1121→    typeof o.keys === "object" &&
  1122→  1122→    o.keys !== null &&
  1123→  1123→    typeof o.policies === "object" &&
  1124→  1124→    o.policies !== null &&
  1125→  1125→    typeof o.backup === "object" &&
  1126→  1126→    o.backup !== null
  1127→  1127→  );
  1128→  1128→}
