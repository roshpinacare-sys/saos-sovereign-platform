   517→function scanProcs(): Array<{ pid: number; cmdline: string; cwd: string }> {
   518→  const out: Array<{ pid: number; cmdline: string; cwd: string }> = [];
   519→  let entries: string[] = [];
   520→  try {
   521→    entries = fs.readdirSync("/proc");
   522→  } catch {
   523→    return out;
   524→  }
   525→  for (const ent of entries) {
   526→    if (!/^\d+$/.test(ent)) continue;
   527→    try {
   528→      const cmdline = fs.readFileSync(`/proc/${ent}/cmdline`, "utf8").replace(/\0/g, " ").trim();
   529→      if (!cmdline) continue;
   530→      let cwd = "";
   531→      try {
   532→        cwd = fs.readlinkSync(`/proc/${ent}/cwd`);
   533→      } catch {
   534→        /* gone / no perms */
   535→      }
   536→      out.push({ pid: Number(ent), cmdline, cwd });
   537→    } catch {
   538→      /* process vanished mid-scan */
   539→    }
   540→  }
   541→  return out;
   542→}

   550→function nextServerRssMb(): { rss: number; ageMs: number } {
   551→  for (const p of scanProcs()) {
   552→    if (p.cmdline.includes("next-server")) {
   553→      try {
   554→        const status = fs.readFileSync(`/proc/${p.pid}/status`, "utf8");
   555→        const m = /VmRSS:\s+(\d+)\s+kB/.exec(status);
   556→        if (m) {
   557→          // v55 — גיל התהליך (שדה 22 ב-/stat = starttime ב-ticks מאז-האתחול):
   558→          // מיחזור-RSS לגיטימי רק על ווב בוגר; ווב צעיר בסערת-קומפילציה נהנה
   559→          // מחלון-חסד (לופ-המיחזור שנתפס חי ב-19:08).
   560→          let ageMs = Infinity;
   561→          try {
   562→            const stat = fs.readFileSync(`/proc/${p.pid}/stat`, "utf8");
   563→            const parts = stat.slice(stat.lastIndexOf(")") + 2).split(" ");
   564→            const startTicks = Number(parts[19]); // field 22 overall (post-comm index 19)
   565→            if (Number.isFinite(startTicks)) {
   566→              const hz = 100;
   567→              const uptimeSec = Number(fs.readFileSync("/proc/uptime", "utf8").split(" ")[0]);
   568→              ageMs = Math.max(0, (uptimeSec - startTicks / hz) * 1000);
   569→            }
   570→          } catch {
   571→            /* age unknown — treat as mature (fail-open to the old doctrine) */
   572→          }
   573→          return { rss: Math.round(Number(m[1]) / 1024), ageMs };
   574→        }
   575→      } catch {
   576→        /* gone */
   577→      }
   578→    }
   579→  }
   580→  return { rss: 0, ageMs: 0 };
   581→}
