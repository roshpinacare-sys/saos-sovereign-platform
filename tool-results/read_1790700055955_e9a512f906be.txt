   195→let lastStateMirrorSig = ""; // `${mtimeMs}:${size}` of the last mirrored dump
   196→function mirrorChainState(): void {
   197→  try {
   198→    const st = fs.statSync(ANVIL_STATE);
   199→    const sig = `${st.mtimeMs}:${st.size}`;
   200→    if (st.size > 0 && sig !== lastStateMirrorSig) {
   201→      fs.mkdirSync(path.dirname(MIRROR_STATE), { recursive: true });
   202→      const dstTmp = `${MIRROR_STATE}.tmp`;
   203→      fs.copyFileSync(ANVIL_STATE, dstTmp);
   204→      // v35 validation — a torn (mid-dump) copy must never reach the home:
   205→      // parse the tmp, promote only a loadable state, else keep the last
   206→      // good mirror and let the next tick retry (anvil dumps every 30s).
   207→      try {
   208→        JSON.parse(fs.readFileSync(dstTmp, "utf8"));
   209→        fs.renameSync(dstTmp, MIRROR_STATE);
   210→        lastStateMirrorSig = sig;
   211→      } catch {
   212→        try { fs.unlinkSync(dstTmp); } catch { /* already gone */ }
   213→        console.log("[evm-mesh] v35 mirror guard: torn anvil-state dump caught — kept the last good mirror");
   214→      }
   215→    }

   240→    fs.statSync(ANVIL_BIN);

   242→    try {
   243→      const st = fs.statSync(MIRROR_BIN);
   244→      if (st.size > 1_000_000) {
   245→        fs.mkdirSync(path.dirname(ANVIL_BIN), { recursive: true });
   246→        fs.copyFileSync(MIRROR_BIN, ANVIL_BIN);
   247→        fs.chmodSync(ANVIL_BIN, 0o755);
   248→        console.log(`[evm-mesh] v34 binary self-heal: anvil ENOENT — restored ${(st.size / 1048576).toFixed(1)}MB from the survivable home (offline, no foundryup)`);
   249→        return;
   250→      }
   251→    } catch {
   252→      // no mirror either — spawn will fail honestly (v33 boundary, now rarer)
   253→    }
   254→    return;
   255→  }
   256→  // the binary answers — keep the mirror armed (size-compare: cheap stat,
   257→  // no 55MB re-hash on every boot; upgrades change the size)
   258→  try {
   259→    const live = fs.statSync(ANVIL_BIN);
   260→    let mirrored = 0;
   261→    try {
   262→      mirrored = fs.statSync(MIRROR_BIN).size;
   263→    } catch {
   264→      mirrored = -1;
   265→    }
   266→    if (mirrored !== live.size) {
   267→      fs.mkdirSync(path.dirname(MIRROR_BIN), { recursive: true });
   268→      const tmp = `${MIRROR_BIN}.tmp`;
   269→      fs.copyFileSync(ANVIL_BIN, tmp);
   270→      fs.renameSync(tmp, MIRROR_BIN);
   271→      console.log(`[evm-mesh] v34 binary mirror armed: ${(live.size / 1048576).toFixed(1)}MB → ${MIRROR_BIN} (the home that survives every rebuild)`);
   272→    }
   273→  } catch {
   274→    // stat race — next boot retries
   275→  }
   276→}

   753→function raceMs<T>(p: Promise<T>, ms: number): Promise<T | null> {
   754→  return new Promise((resolve) => {
   755→    const t = setTimeout(() => resolve(null), ms);
   756→    p.then(
   757→      (v) => { clearTimeout(t); resolve(v); },
   758→      () => { clearTimeout(t); resolve(null); }
   759→    );
   760→  });
   761→}

   763→async function waitTx(tx: ethers.TransactionResponse, ms: number = WAIT_TX_MS): Promise<ethers.TransactionReceipt | null> {
   764→  const t0 = Date.now();
   765→  const hash = tx.hash;
   766→  let stopped = false;
   767→  // רגל-הסקר: בדיקה ראשונה מיידית (automine — הקבלה שם כבר), ואז כל 200ms
   768→  const poller = (async (): Promise<ethers.TransactionReceipt | null> => {
   769→    while (!stopped) {
   770→      const left = ms - (Date.now() - t0);
   771→      if (left <= 0) return null;
   772→      try {
   773→        const rc = await raceMs(tx.provider.getTransactionReceipt(hash), RECEIPT_CALL_MS);
   774→        if (rc) return rc;
   775→      } catch {
   776→        /* קריאה תקועה/נכשלה — מוכרעת ב-2.5ש', הסבב הבא מיד */
   777→      }
   778→      await new Promise((r) => setTimeout(r, Math.min(RECEIPT_POLL_MS, Math.max(0, ms - (Date.now() - t0)))));
   779→    }
   780→    return null;
   781→  })();
   782→  const evWait = tx.wait(1, ms).then(
   783→    (rc) => ({ rc, via: "event" as const }),
   784→    () => ({ rc: null as ethers.TransactionReceipt | null, via: "gate" as const })
   785→  );
   786→  const taggedPoller = poller.then((rc) => ({ rc, via: rc ? ("poll" as const) : ("timeout" as const) }));
   787→  const winner = await Promise.race([evWait, taggedPoller]);
   788→  stopped = true;
   789→  if (winner.via === "poll") {
   790→    // v61 — קטיעת המנוי-הפנימי של ethers על ה-hash: ברגע שהקבלה בידינו,
   791→    // ההמתנה הפנימית של tx.wait (PollingTransactionSubscriber — מנוי-רשת
   792→    // פעיל על ה-hash) היא משקל-מת שממשיך לסקר getTransactionReceipt על
   793→    // כל בלוק עד הדדליין; ובסוף-המחזור provider.destroy() דוחה את הקריאה
   794→    // שבטיסה כ-unhandledRejection שהפילה את הראנר. מפרקים את המנוי מיד —
   795→    // ההמתנה שלנו כבר הוכרעה; ה-waiter הפנימי ייסגר בדדליין שלו בשקט
   796→    // (ה-handlers שלנו על evWait קיימים). בראנר כל hash הוא יחידאי — אין
   797→    // מאזינים אחרים לפרק.
   798→    try {
   799→      tx.provider.removeAllListeners(tx.hash);
   800→    } catch {
   801→      /* provider כבר מת/המנוי כבר פורק */
   802→    }
   803→    console.log(
   804→      `[evm-mesh] v59 receipt-race: ${hash.slice(0, 14)}… rescued by active poll (+${Date.now() - t0}ms) — the event-gate was silent, the chain had the receipt`
   805→    );
   806→  } else if (winner.via === "gate") {
   807→    console.log(
   808→      `[evm-mesh] v56 wait-deadline: ${hash.slice(0, 14)}… stalled >${ms}ms — honest skip (silent wait-gate; the poll saw no receipt either)`
   809→    );
   810→  }
   811→  return winner.rc;
   812→}

  1763→// ─── v36 אכסניית-המושב-היחיד (SINGLE SEAT) ─────────────────────────────────
  1764→// Root fix for the live double-boot incident: /start.sh re-ran over live
  1765→// services and the second mesh runner ADOPTED the live chain — silently
  1766→// doubling every 120s cycle for hours (two runners, one chain: double LLM,
  1767→// double orders, double blooms), because the v20.1 adoption logic was built
  1768→// for an orphaned-chain recovery, not for a second full runner. The seat is
  1769→// a PID lockfile in state/: a second instance exits CLEANLY when a live
  1770→// owner actually serves (its port answers — the zombie lesson: alive ≠
  1771→// serving), takes over a stale seat (dead owner), replaces a broken zombie
  1772→// (owner alive, port dead), and always keeps its own PID (bun --hot
  1773→// re-eval). Fail-safe direction: at most an honest "not starting", never a
  1774→// duplicate runner on the sovereign chain.
  1775→function acquireSeat(tag: string, probePort: number): void {
  1776→  const stateDir = path.join(import.meta.dir, "..", "state");
  1777→  const lockPath = path.join(stateDir, "seat.lock");
  1778→  try { fs.mkdirSync(stateDir, { recursive: true }); } catch { /* exists */ }
  1779→  const release = (): void => {
  1780→    try { if (fs.readFileSync(lockPath, "utf8").trim() === String(process.pid)) fs.rmSync(lockPath, { force: true }); } catch { /* not ours */ }
  1781→  };
  1782→  const GS = globalThis as unknown as { __seatExitBound?: boolean };
  1783→  if (!GS.__seatExitBound) { GS.__seatExitBound = true; process.on("exit", release); }
  1784→  const portAnswers = (): boolean => {
  1785→    try {
  1786→      const r = Bun.spawnSync(["bash", "-c", `exec 3<>/dev/tcp/127.0.0.1/${probePort} 2>/dev/null && exec 3<&- 3>&-`], { timeout: 1500 });
  1787→      return r.exitCode === 0;
  1788→    } catch { return false; }
  1789→  };
  1790→  for (let attempt = 1; attempt <= 3; attempt++) {
  1791→    try {
  1792→      const fd = fs.openSync(lockPath, "wx"); // atomic — EEXIST means the seat is taken
  1793→      fs.writeSync(fd, String(process.pid));
  1794→      fs.closeSync(fd);
  1795→      console.log(`[${tag}] seat acquired (pid ${process.pid}) — single-instance doctrine`);
  1796→      return;
  1797→    } catch (e) {
  1798→      const code = (e as { code?: string })?.code;
  1799→      if (code !== "EEXIST") throw e; // real IO error — crash honestly
  1800→      let ownerPid = 0;
  1801→      try { ownerPid = Number(fs.readFileSync(lockPath, "utf8").trim()); } catch { /* torn write */ }
  1802→      if (ownerPid === process.pid) return; // our own seat — bun --hot re-eval
  1803→      let ownerAlive = false;
  1804→      if (Number.isFinite(ownerPid) && ownerPid > 0) { try { process.kill(ownerPid, 0); ownerAlive = true; } catch { /* dead */ } }
  1805→      if (ownerAlive && portAnswers()) {
  1806→        console.log(`[${tag}] seat: pid ${ownerPid} already serves (port ${probePort} answers) — exiting cleanly, no duplicate`);
  1807→        process.exit(0);
  1808→      }
  1809→      console.log(`[${tag}] seat: ${ownerAlive ? `pid ${ownerPid} alive but NOT serving (zombie)` : `pid ${ownerPid} dead (stale)`} — taking over the seat`);
  1810→      try { fs.rmSync(lockPath, { force: true }); } catch { /* race — retry */ }
  1811→    }
  1812→  }
  1813→  console.log(`[${tag}] seat: contention unresolved — exiting cleanly, no duplicate`);
  1814→  process.exit(0);
  1815→}
