   125→// ─── json-rpc helper (budgeted, failure=data) ───────────────────────────────
   126→
   127→interface RpcOutcome<T> {
   128→  ok: boolean;
   129→  result?: T;
   130→  latencyMs: number;
   131→  error?: string;
   132→}
   133→
   134→async function rpc<T>(
   135→  url: string,
   136→  method: string,
   137→  params: unknown[],
   138→  timeoutMs = 8_000
   139→): Promise<RpcOutcome<T>> {
   140→  const t0 = Date.now();
   141→  try {
   142→    const res = await trustedFetch(url, {
   143→      method: "POST",
   144→      headers: { "Content-Type": "application/json" },
   145→      body: JSON.stringify({ jsonrpc: "2.0", method, params, id: 1 }),
   146→      timeoutMs
   147→    });
   148→    const latencyMs = Date.now() - t0;
   149→    if (!res.ok) {
   150→      return { ok: false, latencyMs, error: `HTTP ${res.status}` };
   151→    }
   152→    const j = (await res.json()) as { result?: T; error?: { message?: string } };
   153→    if (j.error) {
   154→      return { ok: false, latencyMs, error: (j.error.message ?? "rpc error").slice(0, 120) };
   155→    }
   156→    return { ok: true, result: j.result, latencyMs };
   157→  } catch (e) {
   158→    return {
   159→      ok: false,
   160→      latencyMs: Date.now() - t0,
   161→      error: e instanceof Error ? e.message.slice(0, 120) : "rpc failed"
   162→    };
   163→  }
   164→}

   196→async function probeYnetLeg(
   197→  url: string,
   198→  blockNum: number,
   199→  expectedTxid: string
   200→): Promise<ChainLeg> {
   201→  const r = await rpc<CondenserBlock>(url, "condenser_api.get_block", [blockNum], 10_000);
   202→  const base: ChainLeg = {
   203→    ok: false,
   204→    block: blockNum,
   205→    blockId: "",
   206→    timestamp: null,
   207→    txid: "",
   208→    txidExpected: false,
   209→    payload: null,
   210→    latencyMs: r.ok ? r.latencyMs : null,
   211→    error: r.ok ? undefined : r.error
   212→  };
   213→  if (!r.ok || !r.result) return base;
   214→  const blk = r.result;
   215→  base.blockId = (blk.block_id ?? "").slice(0, 20);
   216→  base.timestamp = blk.timestamp ?? null;
   217→  for (const tx of blk.transactions ?? []) {
   218→    const tid = tx.transaction_id ?? "";
   219→    if (tid.toLowerCase() === expectedTxid.toLowerCase()) {
   220→      base.txid = tid;
   221→      base.txidExpected = true;
   222→      for (const op of tx.operations ?? []) {
   223→        if (Array.isArray(op) && op.length === 2 && op[0] === "custom_json") {
   224→          const v = op[1] as { json?: string };
   225→          base.payload = v.json ?? null;
   226→        }
   227→      }
   228→      base.ok = true;
   229→      return base;
   230→    }
   231→  }
   232→  // הבלוק נקרא אבל העסקה לא נמצאה — נתון כנה, לא כישלון-רשת
   233→  base.error = "txid לא נמצא בבלוק (נתון כנה)";
   234→  return base;
   235→}

   271→async function probeSkale(): Promise<SovereignProbe> {
   272→  const t0 = Date.now();
   273→  const [chainId, block, gas, bal] = await Promise.all([
   274→    rpc<string>(SKALE_EUROPA_RPC, "eth_chainId", []),
   275→    rpc<string>(SKALE_EUROPA_RPC, "eth_blockNumber", []),
   276→    rpc<string>(SKALE_EUROPA_RPC, "eth_gasPrice", []),
   277→    rpc<string>(SKALE_EUROPA_RPC, "eth_getBalance", [TREASURY_EVM, "latest"])
   278→  ]);
   279→  const latencyMs = Date.now() - t0;
   280→  if (!chainId.ok || !block.ok) {
   281→    return {
   282→      ok: false,
   283→      latencyMs,
   284→      evidence: "",
   285→      error: chainId.error ?? block.error ?? "skale probe failed"
   286→    };
   287→  }
   288→  const cid = parseInt(chainId.result ?? "0x0", 16);
   289→  const blk = parseInt(block.result ?? "0x0", 16);
   290→  const gasWei = gas.ok ? parseInt(gas.result ?? "0x0", 10) : null;
   291→  const balanceWei = bal.ok ? parseInt(bal.result ?? "0x0", 10) : null;
   292→  // מחיר-גז של 100,000 wei = 1e-13 sFUEL ליחידת-גז — ו-sFUEL חינם מברז הרשת:
   293→  // פריסת relay (~1.5M גז) עולה ~0.00015 sFUEL ≈ $0.00.
   294→  const gasNote =
   295→    gasWei !== null
   296→      ? `gasPrice=${gasWei} wei sFUEL (חינם מברז — לא נכס בעל-ערך)`
   297→      : "gasPrice לא נקרא";
   298→  return {
   299→    ok: true,
   300→    latencyMs,
   301→    evidence: `chainId=${cid} · block=${blk.toLocaleString("en-US")} · ${gasNote} · יתרת-האוצר שלנו (0x01Bd…) = ${balanceWei ?? "?"} wei sFUEL — הדלת נפתחת ב-$0.00`
   302→  };
   303→}

   305→async function probeModuleLane(
   306→  id: string
   307→): Promise<SovereignProbe> {
   308→  const t0 = Date.now();
   309→  try {
   310→    let evidence = "";
   311→    let ok = false;
   312→    switch (id) {
   313→      case "aptos-rpc": {
   314→        const res = await trustedFetch("https://fullnode.mainnet.aptoslabs.com/v1", { timeoutMs: 8_000 });
   315→        if (res.ok) {
   316→          const j = (await res.json()) as { chain_id?: number; ledger_version?: string; block_height?: string };
   317→          ok = typeof j.chain_id === "number";
   318→          evidence = `chain_id=${j.chain_id} · גובה=${Number(j.block_height ?? 0).toLocaleString("en-US")} · ledger ${j.ledger_version}`;
   319→        }
   320→        break;
   321→      }
   322→      case "near-rpc": {
   323→        const r = await rpc<{ chain_id?: string; latest_protocol_version?: number; sync_info?: { latest_block_height?: string } }>(
   324→          "https://rpc.mainnet.near.org", "status", [], 8_000);
   325→        ok = r.ok && r.result?.chain_id === "mainnet";
   326→        evidence = r.ok ? `chain=${r.result?.chain_id} · protocol v${r.result?.latest_protocol_version}` : r.error ?? "";
   327→        break;
   328→      }
   329→      case "polkadot-rpc": {
   330→        const r = await rpc<string>("https://rpc.polkadot.io", "chain_getBlockHash", [0], 8_000);
   331→        ok = r.ok && typeof r.result === "string" && r.result.length > 60;
   332→        evidence = r.ok ? `genesis=${(r.result ?? "").slice(0, 18)}… (אומת חי)` : r.error ?? "";
   333→        break;
   334→      }
   335→      case "tezos-rpc": {
   336→        const res = await trustedFetch("https://rpc.tzbeta.net/chains/main/blocks/head/header", { timeoutMs: 8_000 });
   337→        if (res.ok) {
   338→          const j = (await res.json()) as { chain_id?: string; protocol?: string; level?: number };
   339→          ok = typeof j.chain_id === "string";
   340→          evidence = `chain=${j.chain_id} · level=${j.level} · protocol ${(j.protocol ?? "").slice(0, 12)}…`;
   341→        }
   342→        break;
   343→      }
   344→      case "cosmos-rpc": {
   345→        const r = await rpc<{ node_info?: { network?: string; version?: string }; sync_info?: { latest_block_height?: string } }>(
   346→          "https://cosmos-rpc.publicnode.com", "status", [], 8_000);
   347→        ok = r.ok && r.result?.node_info?.network === "cosmoshub-4";
   348→        evidence = r.ok
   349→          ? `network=${r.result?.node_info?.network} · tendermint v${r.result?.node_info?.version} · גובה=${r.result?.sync_info?.latest_block_height}`
   350→          : r.error ?? "";
   351→        break;
   352→      }
   353→      case "eos-rpc": {
   354→        const res = await trustedFetch("https://eos.greymass.com/v1/chain/get_info", {
   355→          method: "POST",
   356→          timeoutMs: 8_000
   357→        });
   358→        if (res.ok) {
   359→          const j = (await res.json()) as { chain_id?: string; head_block_num?: number };
   360→          ok = typeof j.chain_id === "string";
   361→          evidence = `chain_id=${(j.chain_id ?? "").slice(0, 16)}… · גובה=${(j.head_block_num ?? 0).toLocaleString("en-US")}`;
   362→        }
   363→        break;
   364→      }
   365→      case "vara-rpc": {
   366→        const r = await rpc<string>("https://rpc.vara.network", "chain_getBlockHash", [0], 8_000);
   367→        ok = r.ok && typeof r.result === "string" && r.result.length > 60;
   368→        evidence = r.ok ? `genesis=${(r.result ?? "").slice(0, 18)}… (אומת חי · שרשרת ללא-עמלות)` : r.error ?? "";
   369→        break;
   370→      }
   371→    }
   372→    return { ok, latencyMs: Date.now() - t0, evidence, error: ok ? undefined : "לא אומת במדידה זו" };
   373→  } catch (e) {
   374→    return {
   375→      ok: false,
   376→      latencyMs: Date.now() - t0,
   377→      evidence: "",
   378→      error: e instanceof Error ? e.message.slice(0, 120) : "probe failed"
   379→    };
   380→  }
   381→}

   383→async function probeNostr(host: string): Promise<SovereignProbe> {
   384→  const t0 = Date.now();
   385→  try {
   386→    const res = await trustedFetch(`https://${host}/`, { timeoutMs: 8_000 });
   387→    const latencyMs = Date.now() - t0;
   388→    // ריליי strfry עונה ב-HTML קטן ("strfry: a nostr relay") על GET רגיל —
   389→    // ה-ws עצמו דורש לקוח NOSTR; הנוכחות החיה היא האימות.
   390→    const text = await res.text();
   391→    const ok = res.ok && /nostr relay/i.test(text);
   392→    return {
   393→      ok,
   394→      latencyMs,
   395→      evidence: ok ? `ריליי strfry חי — עונה על HTTP · ws זמין ללקוח NOSTR` : `HTTP ${res.status}`,
   396→      error: ok ? undefined : "תשובה לא-צפויה"
   397→    };
   398→  } catch (e) {
   399→    return {
   400→      ok: false,
   401→      latencyMs: Date.now() - t0,
   402→      evidence: "",
   403→      error: e instanceof Error ? e.message.slice(0, 120) : "nostr probe failed"
   404→    };
   405→  }
