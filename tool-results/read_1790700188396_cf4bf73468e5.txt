   186→/** חריגת-שער: כל ניסיון יציאה למארח לא-רשום. הודעה בעברית, fail-closed. */
   187→export class PhishingGateError extends Error {
   188→  constructor(msg: string) {
   189→    super(msg);
   190→    this.name = "PhishingGateError";
   191→  }
   192→}

   197→/**
   198→ * שער הפישינג — בדיקת URL לפני כל יציאה. מארח חייב להיות ברישום.
   199→ * נחסמים: פרוטוקול לא-https (חוץ מ-loopback), userinfo מוטמע
   200→ * (user:pass@host — טריק פישינג קלאסי), כתובות פרטיות (SSRF),
   201→ * פורטים לא-סטנדרטיים. מחזיר URL מנוקה.
   202→ */
   203→export function assertTrustedUrl(raw: string, context = ""): URL {
   204→  let u: URL;
   205→  try {
   206→    u = new URL(raw);
   207→  } catch {
   208→    throw new PhishingGateError(`שער הפישינג: URL לא-תקין ${context}`);
   209→  }
   210→  if (u.protocol !== "https:" && u.protocol !== "http:") {
   211→    throw new PhishingGateError(`שער הפישינג: פרוטוקול אסור ${u.protocol} ${context}`);
   212→  }
   213→  if (u.username || u.password) {
   214→    throw new PhishingGateError(`שער הפישינג: userinfo מוטמע ב-URL (טריק פישינג) ${context}`);
   215→  }
   216→  const loopback = u.hostname === "127.0.0.1" || u.hostname === "localhost";
   217→  if (!loopback) {
   218→    if (u.protocol !== "https:") {
   219→      throw new PhishingGateError(`שער הפישינג: https בלבד מחוץ למעבדה ${context}`);
   220→    }
   221→    if (u.port && u.port !== "443" && u.port !== "80") {
   222→      throw new PhishingGateError(`שער הפישינג: פורט לא-סטנדרטי :${u.port} ${context}`);
   223→    }

   225→      throw new PhishingGateError(`שער הפישינג: כתובת פרטית חסומה (SSRF) ${u.hostname}`);
   226→    }
   227→  }
   228→  const t = HOST_INDEX.get(u.hostname);
   229→  if (!t) {
   230→    throw new PhishingGateError(
   231→      `שער הפישינג: מארח לא-רשום "${u.hostname}" — היציאה נחסמה (fail-closed). רישום מפורש מהארכיטקט בלבד. ${context}`
   232→    );
   233→  }
   234→  return u;
   235→}

   238→ * fetch מוגן: שער-פישינג + redirect:"error" (לא עוקבים אחרי הפניות —
   239→ * הפניה למארח זר היא וקטור פישינג מוכר) + timeout. כל היציאות
   240→ * החיות של הפלטפורמה עוברות כאן.
   241→ */
   242→export async function trustedFetch(
   243→  raw: string,
   244→  init?: RequestInit & { timeoutMs?: number }
   245→): Promise<Response> {

   252→      ...init,
   253→      redirect: "error",
   254→      signal: ctl.signal
   255→    });
   256→  } finally {
   257→    clearTimeout(t);
   258→  }
   259→}

   402→/** סריקת תוכן-טקסט אחד → ממצאים (בלי תוכן הסוד). */
   403→function scanText(
   404→  text: string,
   405→  file: string,
   406→  canary: Set<string> | null
   407→): SecFinding[] {
   408→  const out: SecFinding[] = [];
   409→  const push = (kind: string, severity: SecFinding["severity"], note: string, idx: number) => {
   410→    const line = text.slice(0, idx).split("\n").length;
   411→    out.push({ file, line, kind, severity, note });
   412→  };
   413→
   414→  for (const m of text.matchAll(PAT_FULL)) {
   415→    push("pat-full", "high", "טוקן GitHub מלא בטקסט — דלף אמיתי", m.index ?? 0);
   416→  }
   417→  for (const m of text.matchAll(PAT_PREFIX)) {
   418→    const s = m[0];
   419→    if (PAT_FULL.test(s)) continue;
   420→    push("pat-prefix", "info", "קידומת-טוקן (לא-שמישה) — שקיפות בלבד", m.index ?? 0);
   421→  }
   422→  for (const m of text.matchAll(MNEMO_RUN)) {
   423→    const ok = isValidMnemonic(m[0]);
   424→    if (ok) {
   425→      push("mnemonic-valid", "high", `רצף ${m[0].trim().split(/\s+/).length} מילים תקף-BIP39 — מנמוניקה אמיתית בטקסט`, m.index ?? 0);
   426→    } else if (canary && canary.size > 0) {
   427→      // קנרית: שלוש-מילים רצופות מהזרעים החתומים (בדיקה בזיכרון בלבד)
   428→      const norm = ` ${m[0].toLowerCase().replace(/\s+/g, " ")} `;
   429→      const words = m[0].toLowerCase().split(/\s+/);
   430→      for (let i = 0; i + 2 < words.length; i++) {
   431→        if (canary.has(`${words[i]} ${words[i + 1]} ${words[i + 2]}`)) {
   432→          push("seed-canary", "high", "קנרית-זרע: מילים מהמנמוניקה החתומה נמצאו בטקסט — דלף ודאי", m.index ?? 0);
   433→          break;
   434→        }
   435→      }
   436→      void norm;
   437→    }
   438→  }
   439→  if (!SKIP_FILE.test(file)) {
   440→    for (const m of text.matchAll(WIF_RUN)) {
   441→      push("wif-suspect", "medium", "תבנית WIL/מפתח-52 — דורש עין (בלוקים ו-ABI מוחרגים)", m.index ?? 0);
   442→    }
   443→  }
   444→  return out;
   445→}

   458→/** סריקת כל blob-י ההיסטוריה דרך cat-file --batch (זרימה אחת). */
   459→function scanHistoryBlobs(canary: Set<string> | null): Promise<{ blobs: number; scanned: number; findings: SecFinding[] }> {
   460→  return new Promise((resolve, reject) => {
   461→    const child = spawn("git", ["cat-file", "--batch"], { cwd: REPO });
   462→    let out: Buffer = Buffer.alloc(0);
   463→    let fed = false;
   464→    const findings: SecFinding[] = [];
   465→    let blobs = 0;
   466→    let scanned = 0;
   467→    let offset = 0;
   468→    const killTimer = setTimeout(() => {
   469→      try { child.kill("SIGKILL"); } catch { /* already gone */ }
   470→      reject(new Error("history scan timeout"));
   471→    }, 60_000);
   472→
   473→    const parse = () => {
   474→      for (;;) {
   475→        const nl = out.indexOf(0x0a, offset);
   476→        if (nl < 0) return;
   477→        const header = out.subarray(offset, nl).toString("utf8").trim();
   478→        const parts = header.split(/\s+/);
   479→        if (parts.length < 3 || !/^\d+$/.test(parts[2] ?? "")) {
   480→          offset = nl + 1;
   481→          continue;
   482→        }
   483→        const size = Number(parts[2]);
   484→        const end = nl + 1 + size;
   485→        if (out.length < end + 1) return; // חסר תוכן — נמתין לעוד נתונים
   486→        blobs += 1;
   487→        if (parts[1] === "blob") {
   488→          const content = out.subarray(nl + 1, end);
   489→          if (size <= 2 * 1024 * 1024 && isTextBuffer(content)) {
   490→            scanned += 1;
   491→            const sha = parts[0] ?? "blob";
   492→            findings.push(...scanText(content.toString("utf8"), `history:${sha.slice(0, 10)}`, canary));
   493→          }
   494→        }
   495→        offset = end + 1; // מדלג על ה-newline החותם
   496→      }
   497→    };
   498→
   499→    child.stdout.on("data", (d: Buffer) => {
   500→      out = Buffer.concat([out, d]);
   501→      parse();
   502→      // ניקוי זיכרון: מקצץ את מה שכבר נצרך
   503→      if (offset > 4 * 1024 * 1024) {
   504→        out = out.subarray(offset);
   505→        offset = 0;
   506→      }
   507→    });
   508→    child.stderr.on("data", () => { /* רעש — נזנח */ });
   509→    child.on("error", (e) => {
   510→      clearTimeout(killTimer);
   511→      reject(e);
   512→    });
   513→    child.on("close", (code) => {
   514→      clearTimeout(killTimer);
   515→      if (code !== 0) {
   516→        reject(new Error(`cat-file exit ${code}`));
   517→        return;
   518→      }
   519→      resolve({ blobs, scanned, findings });
   520→    });
   521→
   522→    exec("git", ["rev-list", "--all", "--objects"])
   523→      .then((list) => {
   524→        const shas = list
   525→          .split("\n")
   526→          .map((l) => l.trim().split(/\s+/)[0] ?? "")
   527→          .filter((s) => /^[0-9a-f]{40}$/.test(s));
   528→        const uniq = Array.from(new Set(shas));
   529→        for (const s of uniq) child.stdin.write(`${s}\n`);
   530→        child.stdin.end();
   531→        fed = true;
   532→      })
   533→      .catch((e) => {
   534→        clearTimeout(killTimer);
   535→        try { child.kill("SIGKILL"); } catch { /* noop */ }
   536→        reject(e instanceof Error ? e : new Error(String(e)));
   537→      });
   538→    void fed;
   539→  });
   540→}

   632→  const files: string[] = [];
   633→  for (const r of roots) {
   634→    if (!fs.existsSync(r)) continue;
   635→    for (const f of fs.readdirSync(r, { recursive: true, encoding: "utf8" }) as unknown as string[]) {
   636→      if (f.endsWith(".ts") || f.endsWith(".tsx")) files.push(path.join(r, f));
   637→    }
   638→  }
   639→  const hosts = new Set<string>();
   640→  const unknown = new Set<string>();
   641→  const display = new Set<string>();
   642→  let rawFetch = 0;
   643→  for (const fp of files) {
   644→    let text = "";
   645→    try {
   646→      text = fs.readFileSync(fp, "utf8");
   647→    } catch {
   648→      continue;
   649→    }
   650→    for (const m of text.matchAll(URL_RE)) {
   651→      const host = (m[1] ?? "").split(":")[0] ?? "";
   652→      if (!host) continue;
   653→      hosts.add(host);
   654→      if (HOST_INDEX.has(host)) continue;
   655→      if (DISPLAY_SET.has(host)) {
   656→        display.add(host);
   657→        continue;
   658→      }
   659→      unknown.add(host);
   660→    }
   661→    // מניין fetch( גולמיים — מספר שקיפות למעקב (המודולים החיים עברו כבר ל-trustedFetch)
   662→    for (const _m of text.matchAll(/(?<![\w.])fetch\s*\(/g)) {
   663→      rawFetch += 1;
   664→    }
   665→  }
   666→  return {
   667→    files: files.length,
   668→    hosts: Array.from(hosts).sort(),
   669→    unknownHosts: Array.from(unknown).sort(),
   670→    displayHosts: Array.from(display).sort(),
   671→    rawFetchCalls: rawFetch,
   672→    note: "מארחי-תצוגה = קישורי אקספלורר (href) שאינם יעדי fetch; מארח לא-רשום ולא-תצוגה = חריגה אמיתית"
   673→  };
   674→}

   762→// פג-תוקף של 60ש' הריץ מחדש cat-file על כל blob (10-13ש' בכל פתיחת וויו

   776→async function canary(): Promise<Awaited<ReturnType<typeof buildCanary>>> {
   777→  if (canaryCache && Date.now() - canaryCache.at < CANARY_MS) return canaryCache.data;

   783→/** v85 — סריקת-ההיסטוריה במטמון-HEAD: זהה לסריקה מלאה בתנאי-אמת זהים. */
   784→async function historyScanCached(trigrams: Set<string> | null): Promise<Awaited<ReturnType<typeof scanHistoryBlobs>>> {
   785→  let head = "";
   786→  try {
   787→    head = (await exec("git", ["rev-parse", "HEAD"], 8_000)).trim();
   788→  } catch {
   789→    head = ""; // אין HEAD — נסרוק בלי מטמון (כנות עולה ביוקף)
   790→  }
   791→  const tc = trigrams?.size ?? 0;
   792→  if (
   793→    head &&
   794→    historyCache &&
   795→    historyCache.head === head &&
   796→    historyCache.trigramCount === tc &&
   797→    Date.now() - historyCache.at < HISTORY_MS
   798→  ) {
   799→    return historyCache.data;
   800→  }
   801→  const data = await scanHistoryBlobs(trigrams);
   802→  if (head) historyCache = { head, trigramCount: tc, at: Date.now(), data };
   803→  return data;
   804→}
