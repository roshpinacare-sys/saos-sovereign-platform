   115→function tokenOk(req) {
   116→  if (!TOKEN) return false;
   117→  const authz = req.headers.get("authorization") ?? "";

   164→function openSeal(dir) {
   165→  const { enc, meta } = encNames(dir);
   166→  const raw = readFileSync(path.join(dir, enc), "utf8");
   167→  if (!raw.startsWith(MAGIC)) throw new Error("unknown seal format");
   168→  const metaJson = JSON.parse(readFileSync(path.join(dir, meta), "utf8"));
   169→  const payload = Buffer.from(raw.slice(MAGIC.length).replace(/\s+/g, ""), "hex");
   170→  const salt = payload.subarray(0, 16), iv = payload.subarray(16, 28), tag = payload.subarray(28, 44), ct = payload.subarray(44);
   171→  const key = scryptSync(metaJson.pass, salt, 32);
   172→  const d = createDecipheriv("aes-256-gcm", key, iv);
   173→  d.setAuthTag(tag);
   174→  return JSON.parse(Buffer.concat([d.update(ct), d.final()]).toString("utf8"));

   282→  const ct = Buffer.concat([c.update(JSON.stringify(bundle), "utf8"), c.final()]);
   283→  const encText = MAGIC + Buffer.concat([salt, iv, c.getAuthTag(), ct]).toString("hex").replace(/(.{64})/g, "$1\n");

   341→function writeIngestedKey(dir, account, network, wif, role) {
   342→  const open = runVault(dir, "--open");
   343→  if (!open.ok) throw new Error(`vault open failed: ${open.out}`);
   344→  const envPath = path.join(dir, ".env");
   345→  if (!existsSync(envPath)) throw new Error("runtime .env missing after vault open (fail-closed)");
   346→  const lines = readFileSync(envPath, "utf8").split(/\r?\n/).filter((l) => l.trim().length > 0);
   347→  const setVar = (name, val) => {
   348→    const i = lines.findIndex((l) => l.startsWith(name + "="));
   349→    if (i >= 0) lines[i] = `${name}=${val}`; else lines.push(`${name}=${val}`);
   350→  };
   351→  // Task 59 (account-aware persist): the ${NET}_${ROLE}_WIF .env slots belong
   352→  // to the MAIN account ONLY. Persisting a non-main account (tov/ynet/fleet)
   353→  // through the old path OVERWROTE headcorner's live-verified slot with a
   354→  // foreign key — a self-inflicted authority poison that the drift monitor
   355→  // would then report as an external rotation. Non-main keys land in
   356→  // keys/vault.json (per-account, net-tagged) and NEVER in main's .env.
   357→  const envMainAccount = lines.find((l) => l.startsWith("AGENT_ACCOUNT="))?.slice("AGENT_ACCOUNT=".length).trim();
   358→  const isMain = !envMainAccount || envMainAccount === account;
   359→  // Task 41-recovery fix: BLURT must write its OWN suffixed slots — the old
   360→  // ternary collapsed BLURT into "STEEM", so BLURT-derived WIFs corrupted the
   361→  // STEEM_* slots (live authority gates then refused every STEEM broadcast).
   362→  const netTag = network === "HIVE" ? "HIVE" : network === "BLURT" ? "BLURT" : "STEEM";
   363→  if (isMain) {
   364→    setVar(`${netTag}_${role.toUpperCase()}_WIF`, wif);
   365→    writeFileSync(envPath, lines.join("\n") + "\n");

   429→/**
   430→ * v2.4.0 — UNIVERSAL key-payload parser ("חיבור חכם לכל מפתח"):
   431→ * the owner can paste ANY of these formats and the signer figures it out:
   432→ *   - a multi-line key file: "OWNER = 5K…", "ACTIVE: 5J…", section headers STEEM/HIVE
   433→ *   - a single WIF (5…/K…/L…), a master password (P…), or a 64-hex SAOS-ROT1 seed
   434→ * Returns entries for /keys/ingest (network/role are HINTS — the live-chain
   435→ * verify step always has the final word, so a wrong hint can never poison a slot).
   436→ */
   437→function parseKeyPayload(text: string) {
   438→  const roleMap: Record<string, string> = {
   439→    OWNER: "owner", OWNERKEY: "owner",
   440→    ACTIVE: "active", ACTIVEKEY: "active",
   441→    POSTING: "posting", POSTINGKEY: "posting",
   442→    MEMO: "memo", MEMOKEY: "memo"
   443→  };
   444→  const entries: Array<{ network: string | null; role: string | null; wif: string }> = [];
   445→  let sectionNet: string | null = null;
   446→  for (const rawLine of String(text ?? "").split(/\r?\n/)) {
   447→    const line = rawLine.trim();
   448→    if (!line || line.startsWith("#") || line.startsWith("//")) continue;
   449→    const sec = line.match(/^(STEEM|HIVE|BLURT)\s*:?\s*$/i);
   450→    if (sec) { sectionNet = sec[1].toUpperCase(); continue; }
   451→    const kv = line.match(/^([A-Za-z][A-Za-z _-]{0,20}?)\s*[=:]\s*(\S+)\s*$/);
   452→    if (kv) {
   453→      let norm = kv[1].toUpperCase().replace(/[\s_-]+/g, "");
   454→      let net = sectionNet;
   455→      const netM = norm.match(/^(STEEM|HIVE|BLURT)(.+)$/);
   456→      if (netM) { net = netM[1]; norm = netM[2]; }
   457→      const role = roleMap[norm] ?? null;
   458→      const wif = kv[2].trim();
   459→      if (wif.length >= 40 && wif.length <= 120) entries.push({ network: net, role, wif });
   460→      continue;
   461→    }
   462→    const spaced = line.match(/^([A-Za-z]{4,8})\s+([5-9KMP][1-9A-HJ-NP-Za-km-z]{48,})$/);
   463→    if (spaced) {
   464→      const role = roleMap[spaced[1].toUpperCase()] ?? null;
   465→      entries.push({ network: sectionNet, role, wif: spaced[2] });
   466→      continue;
   467→    }
   468→    const bare = line.replace(/\s+/g, "");
   469→    const looksKey = /^[5-9KM][1-9A-HJ-NP-Za-km-z]{48,52}$/.test(bare) || /^[0-9a-fA-F]{64}$/.test(bare) || /^P[5-9KM][1-9A-HJ-NP-Za-km-z]{48,52}$/.test(bare);
   470→    if (looksKey) entries.push({ network: sectionNet, role: null, wif: bare });
   471→  }
   472→  return entries.slice(0, 16);
   473→}

   548→const B58_ALPHABET = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
   549→function b58encode(buf) {
   550→  let n = BigInt("0x" + buf.toString("hex"));
   551→  let out = "";
   552→  while (n > 0n) { out = B58_ALPHABET[Number(n % 58n)] + out; n /= 58n; }
   553→  for (const b of buf) { if (b === 0) out = "1" + out; else break; }
   554→  return out;
   555→}
   556→function wifFromPriv32(priv32, compressed) {
   557→  const tail = compressed ? Buffer.from([0x01]) : Buffer.alloc(0);
   558→  const payload = Buffer.concat([Buffer.from([0x80]), priv32, tail]);
   559→  const checksum = createHash("sha256").update(createHash("sha256").update(payload).digest()).digest().slice(0, 4);
   560→  return b58encode(Buffer.concat([payload, checksum]));
   561→}

   564→ * Derive the SAOS-ROT1 tree for one account and match against LIVE authorities.
   565→ * Tries construction variants (key=seed/msg=label and inverse, chain case,
   566→ * compressed/uncompressed WIF) — the exact generator was ephemeral; we match its
   567→ * OUTPUT on-chain, not its source. Returns matches with the exact verifying WIF.
   568→ */
   569→async function saosRot1Match(seedHex, account) {
   570→  const seedBuf = /^[0-9a-fA-F]{64}$/.test(seedHex) ? Buffer.from(seedHex, "hex") : Buffer.from(seedHex, "utf8");
   571→  const matches = [];
   572→  const chainTokens = { STEEM: ["STEEM", "steem"], HIVE: ["HIVE", "hive"] };
   573→  const roles = ["owner", "active", "posting", "memo"];
   574→  for (const net of ["STEEM", "HIVE"]) {
   575→    let auth;
   576→    try { auth = await chainAuthorities(net, account); } catch { continue; }
   577→    for (const chainTok of chainTokens[net]) {
   578→      for (const role of roles) {
   579→        const label = `SAOS-ROT1/${account}/${chainTok}/${role}`;
   580→        const variants = [
   581→          createHmacSha256(seedBuf, Buffer.from(label, "utf8")), // key=seed, msg=label
   582→          createHmacSha256(Buffer.from(label, "utf8"), seedBuf)  // key=label, msg=seed
   583→        ];
   584→        for (const priv32 of variants) {
   585→          for (const compressed of [true, false]) {
   586→            const wif = wifFromPriv32(priv32, compressed);
   587→            let pub = null;
   588→            try { pub = steem.auth.wifToPublic(wif); } catch { continue; }
   589→            if (pub && auth[role] === pub) matches.push({ account, network: net, role, wif, derived: pub });

   591→            else if (pub && auth[role] === "BLT" + pub.slice(3)) matches.push({ account, network: net, role, wif, derived: "BLT" + pub.slice(3) });
   592→          }
   593→        }
   594→      }
   595→    }
   596→  }
   597→  return matches;
   598→}

   647→// roast 56: failure CLASS is a structural fact the caller needs — a chain that
   648→// ANSWERED with an error refused the tx (nothing broadcast → a retry may build

   652→async function rpcCall(node, method, params) {
   653→  const ctl = new AbortController();
   654→  const t = setTimeout(() => ctl.abort(), 15_000);
   655→  try {
   656→    let json;
   657→    try {
   658→      const res = await fetch(node, {
   659→        method: "POST",
   660→        headers: { "content-type": "application/json" },
   661→        body: JSON.stringify({ jsonrpc: "2.0", method, params, id: 1 }),
   662→        signal: ctl.signal
   663→      });
   664→      json = await res.json();
   665→    } catch (e) {
   666→      const err = new Error(`transport: ${String(e?.message ?? e).slice(0, 120)}`);
   667→      err.code = "TRANSPORT";
   668→      throw err;
   669→    }
   670→    if (json.error) {
   671→      const err = new Error(String(json.error.message ?? "rpc error").slice(0, 160));
   672→      err.code = "CHAIN_REJECTED"; // the chain answered and refused — nothing was broadcast
   673→      throw err;
   674→    }
   675→    return json.result;
   676→  } finally {
   677→    clearTimeout(t);
   678→  }
