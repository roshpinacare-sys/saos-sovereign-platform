   153→// ─── תשתית: JSON over https (agent:false, timeout, סבב-חדש לכל בקשה) ───────
   154→
   155→function httpPostJson(url: string, body: string, timeoutMs = 12_000): Promise<unknown> {
   156→  const u = new URL(url);
   157→  return new Promise((resolve, reject) => {
   158→    const req = https.request(
   159→      {
   160→        hostname: u.hostname,
   161→        path: u.pathname || "/",
   162→        method: "POST",
   163→        headers: { "Content-Type": "application/json", "Content-Length": Buffer.byteLength(body) },
   164→        timeout: timeoutMs,
   165→        agent: false
   166→      },
   167→      (res) => {
   168→        const chunks: Buffer[] = [];
   169→        res.on("data", (c: Buffer) => chunks.push(c));
   170→        res.on("end", () => {
   171→          const data = Buffer.concat(chunks).toString("utf8");
   172→          if (res.statusCode !== 200) {
   173→            return reject(new Error(`HTTP ${res.statusCode} @ ${u.hostname}`));
   174→          }
   175→          try {
   176→            resolve(JSON.parse(data));
   177→          } catch {
   178→            reject(new Error(`bad json @ ${u.hostname}`));
   179→          }
   180→        });
   181→      }
   182→    );
   183→    req.on("timeout", () => req.destroy(new Error(`timeout @ ${u.hostname}`)));
   184→    req.on("error", reject);
   185→    req.write(body);
   186→    req.end();
   187→  });
   188→}

   203→async function fetchPrices(): Promise<PriceSet> {
   204→  const symbols = ["ETHUSDT", "BNBUSDT", "TRXUSDT", "XRPUSDT", "SOLUSDT", "POLUSDT", "AVAXUSDT"];
   205→  const empty: PriceSet = {
   206→    ETH: null, BNB: null, TRX: null, XRP: null, SOL: null, POL: null, AVAX: null,
   207→    source: "—"
   208→  };
   209→  try {
   210→    const res = await trustedFetch(
   211→      `https://api.binance.com/api/v3/ticker/price?symbols=${encodeURIComponent(JSON.stringify(symbols))}`,
   212→      { timeoutMs: 8000 }
   213→    );
   214→    if (!res || !res.ok) return empty;
   215→    const rows = (await res.json()) as Array<{ symbol: string; price: string }>;
   216→    const m = new Map(rows.map((r) => [r.symbol, Number(r.price)]));
   217→    const g = (s: string): number | null => {
   218→      const v = m.get(s);
   219→      return v && v > 0 ? v : null;
   220→    };
   221→    return {
   222→      ETH: g("ETHUSDT"),
   223→      BNB: g("BNBUSDT"),
   224→      TRX: g("TRXUSDT"),
   225→      XRP: g("XRPUSDT"),
   226→      SOL: g("SOLUSDT"),
   227→      POL: g("POLUSDT"),
   228→      AVAX: g("AVAXUSDT"),
   229→      source: "Binance ticker (שוק חי)"
   230→    };
   231→  } catch {
   232→    return empty;
   233→  }
   234→}

   257→/** ETH/BSC: הקבלה החיה — gasUsed × effectiveGasPrice (או gasPrice legacy). */
   258→async function evmReceiptCost(rpcUrl: string, chainId: number, txid: string): Promise<LiveCost> {
   259→  const provider = new ethers.JsonRpcProvider(rpcUrl, chainId, { batchMaxCount: 1 });
   260→  try {
   261→    const receipt = await provider.getTransactionReceipt(txid);
   262→    if (!receipt) return { native: null, measuredHe: "קבלה לא נמצאה (הצומת לא ענה)" };
   263→    const price = receipt.gasPrice ?? 0n; // parsed legacy+1559 gasPrice (fee paid per gas)
   264→    const costWei = receipt.gasUsed * price;
   265→    return {
   266→      native: Number(ethers.formatEther(costWei)),
   267→      measuredHe: `gasUsed ${receipt.gasUsed} × ${(Number(ethers.formatUnits(price, "gwei"))).toFixed(3)} gwei — מהקבלה`
   268→    };
   269→  } catch {
   270→    return { native: null, measuredHe: "צומת לא ענה (כנות — לא נמדד)" };
   271→  } finally {
   272→    provider.destroy();
   273→  }
   274→}

   276→/** SOL: עמלת העסקה ב-lamports מההיסטוריה החיה. */
   277→async function solanaFeeCost(txid: string): Promise<LiveCost> {
   278→  try {
   279→    const connection = new Connection(SOL_RPC, "confirmed");
   280→    const t = await connection.getTransaction(txid, { maxSupportedTransactionVersion: 0 });
   281→    if (!t || t.meta === null) return { native: null, measuredHe: "ההיסטוריה לא הוגשה (גיזום-RPC ציבורי)" };
   282→    const lamports = t.meta.fee;
   283→    return {
   284→      native: lamports / 1e9,
   285→      measuredHe: `${lamports.toLocaleString("en-US")} lamports — מההיסטוריה החיה`
   286→    };
   287→  } catch {
   288→    return { native: null, measuredHe: "צומת לא ענה (כנות — לא נמדד)" };
   289→  }
   290→}

   292→/** TRON: עמלה מ-gettransactioninfobyid (fee ב-SUN, 6 ספרות). */
   293→async function tronFeeCost(txid: string): Promise<LiveCost> {
   294→  try {
   295→    const info = (await httpPostJson(`${TRON_API}/wallet/gettransactioninfobyid`, JSON.stringify({ value: txid }))) as {
   296→      fee?: number;
   297→    };
   298→    if (typeof info.fee !== "number") return { native: null, measuredHe: "fee לא הוגש (צומת לא ענה)" };
   299→    return { native: info.fee / 1e6, measuredHe: `${info.fee.toLocaleString("en-US")} SUN — מהשרשרת` };
   300→  } catch {
   301→    return { native: null, measuredHe: "צומת לא ענה (כנות — לא נמדד)" };
   302→  }
   303→}

   313→    if (typeof fee !== "string") return { native: null, measuredHe: "Fee לא הוגש (צומת לא ענה)" };
   314→    return { native: Number(fee) / 1e6, measuredHe: `${Number(fee).toLocaleString("en-US")} drops — מהעסקה המאומתת` };
   315→  } catch {
   316→    return { native: null, measuredHe: "צומת לא ענה (כנות — לא נמדד)" };
   317→  }
