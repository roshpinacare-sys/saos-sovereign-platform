   117→export async function fetchJson(url: string, timeoutMs = 4000): Promise<Record<string, unknown> | null> {
   118→  try {
   119→    // v24 מגן (runner): the mesh service talks ONLY to the local platform
   120→    // (127.0.0.1) — any other host is blocked before connect (fail-closed,
   121→    // anti-phishing: unknown hosts are never probed from the runner).
   122→    const u = new URL(url);
   123→    if (u.hostname !== "127.0.0.1" && u.hostname !== "localhost") {
   124→      console.error(`[magen] blocked outbound fetch to ${u.hostname} (loopback only)`);
   125→      return null;
   126→    }
   127→    // v55 לולאת-המוות שנתפסה חי (restart #1/#2): בזמן מצוקת-ווב (RSS 1.5GB ·
   128→    // page-hang) הווב מחזיר 200+כותרות אבל הגוף נתקע — ו-bun לעולם לא דוחה
   129→    // את res.json() ב-abort (הוכח בריפרו repro-abort.ts ניסוי B: תלייה>15ש'
   130→    // למרות abort). הטיפול: ה-abort ממשיך לכסות גם את קריאת-הגוף (clearTimeout
   131→    // רק ב-finally), וקריאת-הגוף עצמה רצה במירוץ קשיח נגד deadline שתמיד
   132→    // מסתיים — לא סומכים על ה-abort, ה-fetch-כולו חסום-זמן.
   133→    const ctl = new AbortController();
   134→    const t = setTimeout(() => ctl.abort(), timeoutMs);
   135→    try {
   136→      const res = await fetch(url, { signal: ctl.signal });
   137→      if (!res.ok) return null;
   138→      // v55 — המירוץ הקשיח עם ניקוי-טיימר: המנצח מנקה את ה-deadline (אין
   139→      // לוגי-שקר מאוחרים); המפסיד מוחזר כ-null כנה.
   140→      const body = await new Promise<Record<string, unknown> | null>((resolve) => {
   141→        const t2 = setTimeout(() => {
   142→          console.log(`[evm-mesh] v55 body-deadline: ${u.pathname} stalled >${timeoutMs}ms — honest null (bun abort does not reject a stalled body read)`);
   143→          resolve(null);
   144→        }, timeoutMs);
   145→        res.json().then(
   146→          (v) => { clearTimeout(t2); resolve(v as Record<string, unknown>); },
   147→          () => { clearTimeout(t2); resolve(null); }
   148→        );
   149→      });
   150→      return body;
   151→    } finally {
   152→      clearTimeout(t);
   153→    }
   154→  } catch {
   155→    return null;
   156→  }
   157→}

   159→/** LIVE anchors from the platform's 8-chain grid (best-effort). */
   160→export async function fetchLiveAnchors(): Promise<{
   161→  chains: string[];
   162→  heights: bigint[];
   163→  digests: string[];
   164→} | null> {
   165→  const [probes, status] = await Promise.all([
   166→    fetchJson("http://127.0.0.1:3000/api/saos/live-probes"),
   167→    fetchJson("http://127.0.0.1:3000/api/saos/status")
   168→  ]);
   169→  const h: Record<string, number> = {};
   170→  const stats = status?.stats as { lastHeight?: number } | undefined;
   171→  if (stats?.lastHeight != null) h["SAOS-NET"] = Number(stats.lastHeight);
   172→  const eth = probes?.eth as { blockNumber?: number } | undefined;
   173→  const bsc = probes?.bsc as { blockNumber?: number } | undefined;
   174→  const sol = probes?.sol as { slot?: number } | undefined;
   175→  if (eth?.blockNumber != null) h["ETHEREUM"] = Number(eth.blockNumber);
   176→  if (bsc?.blockNumber != null) h["BSC"] = Number(bsc.blockNumber);
   177→  if (sol?.slot != null) h["SOLANA"] = Number(sol.slot);
   178→
   179→  const chains: string[] = [];
   180→  const heights: bigint[] = [];
   181→  const digests: string[] = [];
   182→  for (const c of MESH_CHAINS) {
   183→    const v = h[c];
   184→    if (v != null && Number.isFinite(v) && v > 0) {
   185→      chains.push(c);
   186→      heights.push(BigInt(v));
   187→      digests.push(ethers.id(`saos:${c}:${v}`));
   188→    }
   189→  }
   190→  return chains.length > 0 ? { chains, heights, digests } : null;
   191→}

   226→export async function fetchRealAssets(): Promise<{ assets: Record<string, RealAsset>; live: number }> {
   227→  const flow = await fetchJson("http://127.0.0.1:3000/api/saos/flow", 25000);
   228→  const rows = (((flow?.radar as { assets?: unknown[] } | undefined)?.assets) ?? []) as Array<{
   229→    symbol?: string;
   230→    holder?: string;
   231→    address?: string;
   232→    balanceNum?: number;
   233→    priceUsd?: number | null;
   234→  }>;
   235→  const pick = (symbol: string, prefix: string): RealAsset | null => {
   236→    const a = rows.find(
   237→      (x) => x.symbol === symbol && x.holder === "headcorner" && typeof x.address === "string" && x.address.startsWith(prefix)
   238→    );
   239→    if (!a || typeof a.balanceNum !== "number" || !(a.balanceNum > 0)) return null;
   240→    if (typeof a.priceUsd !== "number" || !(a.priceUsd > 0)) return null;
   241→    return {
   242→      symbol,
   243→      units: a.balanceNum,
   244→      usd: a.priceUsd,
   245→      source: `מכ"ם חי ${new Date().toISOString().slice(0, 10)}`,
   246→      address: a.address as string
   247→    };
   248→  };
   249→  const out: Record<string, RealAsset> = {};
   250→  let live = 0;
   251→  for (const [sym, prefix] of [
   252→    ["TRX", "TG6Srs"],
   253→    ["SOL", "9knfnF"],
   254→    ["ETH", "0x2767"]
   255→  ] as const) {
   256→    const got = pick(sym, prefix);
   257→    if (got) {
   258→      out[sym] = got;
   259→      live += 1;
   260→    } else {

   264→  }

   284→async function sendTx(label: string, call: () => Promise<ethers.ContractTransactionResponse>): Promise<ethers.ContractTransactionReceipt | null> {
   285→  const tx = await call();
   286→  // v91 receipt-poll fix (measured 2026-09-13): tx.wait() hung forever even
   287→  // though the tx DID mine (block 11 carried the hub.wire selector, zero "✓"
   288→  // lines, 1019 blockNumber heartbeats, empty txpool) — the ethers polling
   289→  // transaction-subscriber race from the v61 Promise-Guard family. We no
   290→  // longer blindly trust wait(): poll the chain for the receipt ourselves
   291→  // and fail LOUDLY on timeout so the watchdog loop recovers honestly.
   292→  let rc: ethers.ContractTransactionReceipt | null = null;
   293→  for (let i = 0; i < 240 && !rc; i++) {
   294→    try {
   295→      rc = (await tx.provider.getTransactionReceipt(tx.hash)) as ethers.ContractTransactionReceipt | null;
   296→    } catch {
   297→      // transient RPC hiccup — keep polling, the chain is the truth
   298→    }
   299→    if (!rc) await new Promise((r) => setTimeout(r, 250));
   300→  }
   301→  if (!rc) {
   302→    console.error(`[mesh] sendTx ${label}: no receipt in 60s for ${tx.hash} — fail-loud (watchdog recovers)`);
   303→    throw new Error(`sendTx ${label}: receipt timeout for ${tx.hash}`);
   304→  }
   305→  console.log(`[mesh] ✓ ${label} (block ${rc.blockNumber})`);
   306→  return rc;
   307→}

   321→export async function deployMesh(): Promise<DeploymentState> {
   322→  const artifact = loadArtifact();
   323→  fs.mkdirSync(path.dirname(STATE_PATH), { recursive: true });
   324→
   325→  // v54 מכוון-הרשת — ריעוף 120ms לפריסה/המתנות
   326→  const provider = new ethers.JsonRpcProvider(MESH_RPC, MESH_CHAIN_ID, { batchMaxCount: 1, pollingInterval: 120 });
   327→  const operator = await provider.getSigner(0);
   328→  const operatorAddr = await operator.getAddress();
   329→
   330→  const abiOf = (name: string): ethers.InterfaceAbi => {
   331→    const c = artifact.contracts.find((x) => x.name === name);
   332→    if (!c) throw new Error(`artifact missing ${name}`);
   333→    return c.abi;
   334→  };
   335→
   336→  // 1. deploy all seven contracts
   337→  const addr: Record<string, string> = {};
   338→  for (const c of artifact.contracts) {
   339→    const factory = new ethers.ContractFactory(c.abi, c.bytecode, operator);
   340→    const deployed = await factory.deploy();
   341→    await deployed.waitForDeployment();
   342→    addr[c.name] = await deployed.getAddress();
   343→    console.log(`[mesh] deployed ${c.name} → ${addr[c.name]} (${c.bytecodeBytes}B)`);
   344→  }
   345→
   346→  const C = (name: string) => new ethers.Contract(addr[name] as string, abiOf(name), operator);
   347→  const hub = C("SAOSMeshHub") as unknown as HubContract;
   348→  const relay = C("SAOSMeshRelay");
   349→  const identity = C("SAOSIdentity");
   350→  const ledger = C("SAOSLedger") as unknown as LedgerContract;
   351→  const solutions = C("SAOSSolutions");
   352→  const bridge = C("SAOSBridge");
   353→  const workboard = C("SAOSWorkBoard");
   354→  const microgrid = C("SAOSMicrogrid");
   355→  const exchange = C("SAOSExchange") as unknown as ExchangeContract;
   356→  const value = C("SAOSValueEngine");
   357→
   358→  // v4 — אוצר-הבורסה / reserve custodian: חותם 9 (anvil unlocked). כל הקציר
   359→  //      וההון-יוצר-השוק של הבורסה יושבים כאן — כתובת פומבית בלבד, אין מפתח.
   360→  const treasurySigner = await provider.getSigner(9);
   361→  const treasuryAddr = await treasurySigner.getAddress();
   362→
   363→  // 2. wire the chain — every arrow below is a real contract-to-contract link
   364→  await sendTx("hub → relay + valueEngine", () => hub.wire(addr.SAOSMeshRelay, addr.SAOSValueEngine));
   365→  await sendTx("relay → hub", () => relay.wireHub(addr.SAOSMeshHub));
   366→  for (const link of LINK_ORDER) {
   367→    await sendTx(`relay link + ${link}`, () => relay.addLink(addr[link] as string));
   368→  }
   369→  await sendTx("identity → relay", () => identity.wire(addr.SAOSMeshRelay as string));
   370→  await sendTx("ledger → relay", () => ledger.wire(addr.SAOSMeshRelay as string));
   371→  await sendTx("solutions → relay", () => solutions.wire(addr.SAOSMeshRelay as string));
   372→  await sendTx("bridge → relay", () => bridge.wire(addr.SAOSMeshRelay as string));
