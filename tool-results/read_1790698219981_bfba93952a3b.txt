    51→/** מחיר-הוגן: עדות-משוקללת של הספר, הבריכה וסולם-הערך */
    52→export function fairPrice(st: DexState, base: string, quote: string): number {
    53→  const ref = st.assets[base]?.refPrice ?? 0;
    54→  // R26 · עוגן-יחיד לזוגות-המסילה: הפייר של זוג-בבעלות-המסילה הוא refPrice — לא
    55→  // תערובת-ספר+בריכה. המצאו-החי (h18553): 80% מהפייר היה self-referential
    56→  // (ספר 50% + בריכה 30%) — נקודת-שבת-עצמית שהחזיקה-את-הספר ב-305µ בעוד-העוגן
    57→  // 54µ (STEEM-אמת). "מנוע-אחד-לעוגן" (SETTLE-16) — המסילה היא בעלת-המשמעות,
    58→  // והעוגן שלה הוא-האורקל, תמיד. הספר מצטט-סביב-העוגן ± spread — נקודה.
    59→  if (ref > 0 && RAIL_OWNED_POOLS.has(poolKey(base, quote))) return ref;
    60→  const bids = sortBook(st, base, quote, "buy");
    61→  const asks = sortBook(st, base, quote, "sell");
    62→  const bestBid = bids[0]?.price ?? 0;
    63→  const bestAsk = asks[0]?.price ?? 0;
    64→  const bookMid = bestBid > 0 && bestAsk > 0 ? Math.floor((bestBid + bestAsk) / 2) : (bestBid || bestAsk || 0);
    65→  const p = getPool(st, poolKey(base, quote));
    66→  const pm = p ? poolMidFor(p, base, quote) : 0;
    67→  // משקלים: ספר 50% · בריכה 30% · סולם-ערך 20% (נפילה-חן למקורות-קיימים)
    68→  const parts: [number, number][] = [];
    69→  if (bookMid > 0) parts.push([bookMid, 0.5]);
    70→  if (pm > 0) parts.push([pm, 0.3]);
    71→  if (ref > 0) parts.push([ref, 0.2]);
    72→  if (parts.length === 0) return ref;
    73→  const wsum = parts.reduce((s, [, w]) => s + w, 0);
    74→  return Math.floor(parts.reduce((s, [v, w]) => s + v * w, 0) / wsum);
    75→}

    77→function skewShiftBps(st: DexState, soldier: string, base: string, params: ArenaParams): number {
    78→  // FIX(SETTLE-7): קראו מ-params החיים (לא DEFAULT_PARAMS) — שינוי-פרמטרים ב-API
    79→  // לא-השפיע על ה-skew והטלמטריה סטתה מהגריד בפועל.
    80→  const inv = st.balances[soldier]?.[base] ?? 0;
    81→  const target = Math.floor(params.maxInventoryMu / 2);
    82→  const dev = (inv - target) / params.maxInventoryMu; // -0.5..0.5+
    83→  const clamped = Math.max(-1, Math.min(1, dev * 2));
    84→  return Math.floor(clamped * (params.spreadBps / 2)); // מלאי-כבד → מוכר-זול יותר
    85→}

   153→/** FIX(Task 56 replay-storm): אחרי-טעינת-תמונת-מצב, st.nonces מכיל אלפי-נונסים
   154→ * `a<N>` של-החיילים-מהריצות-הקודמות; RT-רץ-חדש התחיל מ-seq=0 והתנגש-בכולם
   155→ * (replay-rejected המוני עד-שה-seq עובר-את-החלון). הפתרון: בעת-יצירת-RT לחייל,
   156→ * נזרע-seq מעל-המקסימום-הקיים-בחלון-הנונסים — התנגשות-אפסית מהטיק-הראשון. */
   157→function seedNonceSeq(st: DexState, soldier: string): number {
   158→  let max = 0;
   159→  const prefix = `${soldier}:a`;
   160→  for (let i = st.nonces.length - 1; i >= 0 && i >= st.nonces.length - 2000; i--) {
   161→    const n = st.nonces[i];
   162→    if (n.startsWith(prefix)) {
   163→      const v = parseInt(n.slice(prefix.length), 10);
   164→      if (Number.isFinite(v) && v > max) max = v;
   165→    }
   166→  }
   167→  return max;
   168→}

   200→  const m = MARKETS[0]; // הזוג-הריבוני — לחגורת-הסולם
   201→  const fair = fairPrice(st, m.base, m.quote);

   204→  // (1) מנעול-בריכה — לכל-הזוגות (FIX SETTLE-14: היה רק-הזוג-הריבוני ובריכת

   206→  // R26 · הסרת-הפטור-של-זוגות-המסילה: הפטור נולד (SETTLE-16) כי-הפייר-הצף גרר-

   209→  // מנוע-אחד, יעד-אחד (העוגן). המנעול חוזר-לשמור-גם-על-בריכת-המסדרון — בלי-
   210→  // הכוח-הזה בריכה-רדודה (אחרי-סימון-מחדש) נסחפת-חופשית-ברעש-הבוטים (נמדד:
   211→  // 54→66µ ב-60-טיקים) והמסילה חסרת-מלאי-לשפשף (doses=0). המנוע-השני נשאר
   212→  // למנות-העדינות (TWAP-scrubbing) — שניהם-פונים-לאותו-עוגן.
   213→  for (const mk of MARKETS) {
   214→    const mkPair = poolKey(mk.base, mk.quote);
   215→    const p = getPool(st, mkPair);
   216→    if (!p) continue;
   217→    const pm = poolMidFor(p, mk.base, mk.quote);
   218→    if (pm <= 0) continue;
   219→    const mkFair = fairPrice(st, mk.base, mk.quote); // פייר-פר-זוג — לעולם לא פייר-זר
   220→    if (mkFair <= 0) continue;
   221→    const gapBps = Math.floor(((pm - mkFair) / mkFair) * 10_000);
   222→    // SETTLE-14: פער-גדול (≥200bps) מטופל כל-טיק; קטן — כל-טיק-שני (חיסכון-גז)
   223→    if (Math.abs(gapBps) >= 30 && (Math.abs(gapBps) >= 200 || tick % 2 === 0)) {
   224→      const assetIn = gapBps > 0 ? mk.base : mk.quote; // יקר-בבריכה → מזרימים-בסיס-פנימה (מוריד-mid)
   225→      // R26 · מנה-מדויקת-שורש (סגירת-פער-לאפס): הליניאריזציה-מחצית-הפער היתה
   226→      // כפולה במכפיל — mid ∝ ra/rb (שינוי-יחס), אבל המנה חושבה-ליניארית-על-המונה
   227→      // בלבד. על-בריכה-רדודה מנת-10kµ הרימה-את-ה-mid ב-+3.7% (דיוק-היעד: +1.9%)
   228→      // → הצד-השני לא-החזיר-אותה → ראצ'ט מתמשך שדילדל-את-הבריכה מ-WSTEEM
   229→      // (נמדד-חי: 54→82µ תוך-240-טיקים). המנה-המדויקת להביא-את-מחיר-הבריכה
   230→      // בדיוק-לפייר: sqrt — שכן mid' = 1000·rQ·rB/(rB+x)² למכירה,
   231→      // ו-mid' = 1000(rQ+x)²/(rB·rQ) לקנייה. לפערים-קטנים היא מתכנסת-לליניארית
   232→      // (sqrt(1+g)≈1+g/2) — שיפור-טהור, בלי-overshoot, בלי-ראצ'ט.
   233→      const rBase = mk.base === p.a ? p.ra : p.rb;
   234→      const rQuote = mk.quote === p.a ? p.ra : p.rb;
   235→      const raw = gapBps > 0
   236→        ? Math.floor(Math.sqrt(Math.floor((1000 * rQuote * rBase) / mkFair)) - rBase)   // מכירת-בסיס פנימה
   237→        : Math.floor(Math.sqrt(Math.floor((mkFair * rBase * rQuote) / 1000)) - rQuote); // קניית-בסיס מהבריכה
   238→      const size = Math.min(Math.max(raw, 1), 600_000, avail(st, "arena", assetIn));
   239→      if (size >= 500) {
   240→        pushOp({ type: "pool_swap", issuer: "arena", nonce: `pk-pool-h${st.height}-${mk.base}`, pair: poolKey(mk.base, mk.quote), assetIn, amount: size });
   241→      }
   242→    }
   243→  }
   244→
   245→  // (2) חגורת-סולם: bookMid ∈ ref±400bps
   246→  const ref = st.assets[m.base]?.refPrice ?? 0;
   247→  if (ref > 0 && tick % 4 === 0) {
   248→    const bb = sortBook(st, m.base, m.quote, "buy")[0]?.price ?? 0;
   249→    const ba = sortBook(st, m.base, m.quote, "sell")[0]?.price ?? 0;
   250→    const mid = bb > 0 && ba > 0 ? Math.floor((bb + ba) / 2) : (bb || ba);
   251→    if (mid > 0) {
   252→      const devBps = Math.floor(((mid - ref) / ref) * 10_000);
   253→      // SETTLE-14 price-truth: גודל-החגורה פרופורציונלי לסטייה — 80k-קבוע מול
   254→      // סטיית-15% היה טיפה-בים (הספר צף ולא חזר). 120k+300µ/סט-בפס, תקרה 600k.
   255→      const bandQty = Math.min(120_000 + Math.abs(devBps) * 300, 600_000);
   256→      if (devBps > 400) {
   257→        // יקר-מדי-מול-הסולם → הארנה מוכרת-לתוך-הבידים (מורידה-את-המגע)
   258→        const qty = Math.min(bandQty, avail(st, "arena", m.base));
   259→        if (qty >= 5_000) {
   260→          pushOp({ type: "fill", issuer: "arena", nonce: `pk-band-h${st.height}-${m.base}-s`, base: m.base, quote: m.quote, side: "sell", amount: qty, market: true });
   261→        }
   262→      } else if (devBps < -400) {
   263→        // זול-מדי → הארנה קונה-מהאסקים (מרימה-את-המגע)
   264→        const px = ba > 0 ? ba : ref;
   265→        const qty = Math.min(bandQty, Math.floor((avail(st, "arena", m.quote) * 900) / (px * 10)));
   266→        if (qty >= 5_000) {
   267→          pushOp({ type: "fill", issuer: "arena", nonce: `pk-band-h${st.height}-${m.base}-b`, base: m.base, quote: m.quote, side: "buy", amount: qty, market: true });
   268→        }
   269→      }
   270→    }
   271→  }
   272→
   273→  // (3) רצפות-עומק (SETTLE-14): "הבריכות יהיו זמינות" — בריכה-שרזרבה-אחת-שלה
   274→  // מתחת-לרצפה מקבלת-השלמה-פרופורציונלית-מהארנה (הוספה-ביחס-הנוכחי = אפס-הזזת-mid).
   275→  // ממומן-רק-ממלאי-אמת-של-הארנה (בדיקת-avail לפני-הדחיפה — אפס-דחיות-ספר).
   276→  if (tick % 8 === 0) {
   277→    for (const m of MARKETS) {
   278→      const p = getPool(st, poolKey(m.base, m.quote));
   279→      if (!p) continue;
   280→      const floorA = POOL_DEPTH_FLOORS[p.a] ?? 0;
   281→      const floorB = POOL_DEPTH_FLOORS[p.b] ?? 0;
   282→      const shortA = Math.max(0, floorA - p.ra);
   283→      const shortB = Math.max(0, floorB - p.rb);
   284→      if (shortA <= 0 && shortB <= 0) continue;
   285→      // משלימים-את-הצד-החסר-הגדול; הצד-השני נגזר-מיחס-הבריכה (חוזה pool_add)
   286→      const addA = shortA >= shortB ? shortA : (p.ra > 0 ? Math.floor((shortB * p.ra) / p.rb) : 0);
   287→      if (p.ra <= 0 || p.rb <= 0 || addA <= 0) continue;
   288→      const addB = Math.floor((p.rb * addA) / p.ra);
   289→      if (addA < 20_000 || addB < 20_000) continue;
   290→      if (avail(st, "arena", p.a) < addA || avail(st, "arena", p.b) < addB) continue;
   291→      pushOp({ type: "pool_add", issuer: "arena", nonce: `pk-floor-h${st.height}-${p.a}`, pair: poolKey(p.a, p.b), amount: addA });
   292→    }
   293→  }
   294→}

   296→/** טיק-ארנה: ריקוט-גריד לכל-חייל + טייק-מסתובב עם-שער-שוליים + בוטי-רעש
   297→ * FIX(SETTLE-9): ה-param nowMs היה-מת (void nowMs בלבד) — הוסר מהחתימה ומהקריאה. */

   307→  pegkeeperTick(st, tick, pushOp);

   310→  // הקיים (אפס-הטבעה) + מנות-שפשוף-לעוגן על הזוגות-בבעלותה. אחרי-pegkeeper,

   315→  // הזרמה מהאוצר. **העברה-טהורה (אפס-הטבעה)** — Σcredits נשמר. בלי-זה הגריד

   317→  // הספרד-נעלמת והשוק מפסיק-לשבת. תיקון-שורש לתשישות-הכלכלה-הסימולטיבית.

   321→  // ל-4 (וגם-טיק 1–2 מיד-אחרי-boot): התזמון בלבד — ההעברות-עצמן טהורות-ומוגבלות.

   323→    // FIX(roast-81 · afford-race): ספר-תקציב-פר-טיק (ראה-למטה) — אפס-חפירה-מעבר-לרצפת.

   326→    // feeMeter-מעל-רצפת-הסולם, מדוד-תקרה) — המסחר-עצמו מחזיר-את-הדלק-ששרף.
   327→    // קודם-ההטבעה, אחר-כך העירוי.

   329→    // (מתחת-ל-20k) = מזרקה-יבשה-לנצח בזמן-שלטרשורה-היה-גיבוי-אמת. השער הורד-ל-1k:
   330→    // הגרעין (kernel mint_fuel) עדיין אוכף בעצמו את הגיבוי-הכפול, הרצפה (8k) והתקרה —

   334→    // (mint_fuel כבר-אוכף-בעצמו רצפת-8k + תקרה). נמדד-חי: fuelAvail=577<שער-1,000
   335→    // בזמן-שהגרעין-הרשה-1,154 → מזרקה-יבשה, האוצר-תקוע-על-8µ-גז, ו-ProtoRev —
   336→    // מנוע-הארב-שרץ-כ-batch-אוצר — מת-מרעב-גז (revAux.rejections=5,181).
   337→    // עכשיו: אותו-מדד-בדיוק-של-הגרעין — האכיפה-במקום-אחד.

   339→    if (trCr < 10_000 && fuelAvail > 1_000) {

   344→      const c = st.credits[ag] ?? 0;
   345→      // FIX(SETTLE-15 fuel-oscillation, נתפס-בלוג-הדחיות): סף-10k = מילוי-מוקדם-בחלון-אחד,
   346→      // אפס-דחיות; זו-העברה-טהורה (אפס-הטבעה) — השריפה-הכוללת-לא-משתנה, רק-התזמון-מוקדם.
   347→      if (c < 10_000) {
   348→        const need = Math.min(30_000 - c, REFUEL_CAP);
   349→        if (need <= 0) continue;
   350→        // FIX(roast-81 · affordability-ask): היה-בקשה-קבועה (30k−c) שהגרעין דוחה-בכנות
   351→        // כשאף-ממנה לא עומד-במרווח — נמדד-חי: Σcredits=54,793µ מול-דרישה-קבועה-של-~240k
   352→        // → סירול-דחיות-נצחי, הספר-ריק-מציטוטים, וחיילים-עשירים-בטוקנים-ועניים-בגז
   353→        // (soldier-1: 2.1M-SAOS+3.8M-USDS · 10µ-קרדיטים) לא-מצטטים. עכשיו: הבקשה
   354→        // מתכווננת-ליכולת-הממנה-בפועל — אותו-סדר-סולם-ואותם-מרווחי-ברזל (funderAfford
   355→        // מקור-אמת-אחד בגרעין); מתחת-ל-FUEL_ASK_MIN לא-מבקשים (אבק-גז); אסור-לממן-עצמי
   356→        // (f≠ag — SETTLE-7). הגרעין נשאר-האוכף-המלא (fail-closed).
   357→        // FIX(roast-81 · afford-race): כל-הדחיפות-באותו-טיק קוראות-את-אותו-st.credits
   358→        // לפני-היישום — נמדד-חי: bot-a (14,940) מומן-3×4,940-בטיק-אחד ונשחק-עד-0
   359→        // (עבר-על-רצפת-ה-10k-שלו). עכשיו: ספר-תקציב-פר-טיק — כל-דחיפה-מנכה-מהיכולת
   360→        // שנתפסה, אף-ממנה-לא-נחפר-מעבר-לרצפת-שלו.
   361→        let funder: string | null = null;

   366→          if (afford >= Math.min(need, FUEL_ASK_MIN)) { funder = f; amt = Math.min(need, afford); break; }
   367→        }
   368→        if (!funder || amt <= 0) continue;
   369→        fuelBudget.set(funder, (fuelBudget.get(funder) ?? 0) + amt);

   386→      const amt = Math.min(REARM_TRANCHE_USDS, REARM_LIFETIME_USDS - rc.usds);
   387→      if (avail(st, "treasury", "USDS") >= amt) {
   388→        rc.usds += amt;
   389→        pushOp({ type: "transfer", issuer: "treasury", nonce: `rearm-uh${st.height}`, to: "arena", asset: "USDS", amount: amt });
   390→      }
   391→    }
   392→    // קו-WSTEEM (SETTLE-14): אוצר-עוטף-את-ה-STEEM-הטבעי-שלו-עצמו (גיבוי 1:1 כנה —
   393→    // אפס-הטבעה-רכה: הגיבוי נרשם ב-backing) ואז מעביר לארנה לשירות-המנעול.
   394→    // העטיפה-וההעברה-בטיקים-נפרדים — הבדיקה-על-avail לפני-כל-דחיפה מסדרת-את-הסדר.

   397→      const tw = avail(st, "treasury", "WSTEEM");
   398→      if (tw < amt && avail(st, "treasury", "STEEM") >= amt - tw) {
   399→        pushOp({ type: "wrap_mint", issuer: "treasury", nonce: `rearm-wraph${st.height}`, wrapAsset: "WSTEEM", amount: amt - tw });

   408→    // באותו-חלון-16-טיק של-ה-rearm: קודם-הארנה-מתמלאת, אחר-כך-החיילים-מקבלים.

   417→      const amt = Math.min(want, byCap, arenaSpare);

   420→        pushOp({ type: "transfer", issuer: "arena", nonce: `swseed-h${st.height}-${soldier}`, to: soldier, asset: "WSTEEM", amount: amt });

   425→  // קצבת-בוטים: כל-24 טיקים ממלאים-מחדש סוחרי-רעש שנדלדלו — בלי-זה הזרימה-
   426→  // החופשית מתה (אפס-מילויים → אפס-גילוי-מחיר). **המפעיל-הנדיב: הארנה**
   427→  // (המייצבת-עם-המלאי-העמוק) — לא-האוצר: לאוצר יש רק-את-קיזוז-העמלות,
   428→  // והקצבה-ממנו-רוקנה-אותו-עד-ריק (ממצא-חי 2026-09). מדידה-על-**avail** (כולל
   429→  // נעילות-ספר) ולא-יתרה-גולמית — פקודות-מנוחות של-בוט לא-ימנעו-מימון-חדש.
   430→  if (params.noise && tick % 24 === 0) {
   431→    // R28 · bot-e = סוכן-ההתעמלות (ראה exerciseTick) — מקבל את אותה קצבת-בוטים
   432→    // מהארנה ואותן תקרות-חיים (2.4M USDS / 1.2M SAOS): סוחר-רעש לכל דבר, רק

   674→  // מדידת-מלאי R28 (בדיקת-ההוכחה המקומית): בוטי-הרעש רוקנו עידנים-עידנים
   675→  // (SAOS=1µ — תקרות-הזריעה מוצו), החיילים חסרי-SAOS — הצד-השני היחיד עם
   676→  // כיסים-עמוקים משני-הצדדים ועם התפקיד-הארכיטקטוני המתאים הוא הארנה —
   677→  // "המפעיל-הנדיב" שמממן את הסימולציה מלכתחילה (קצבות, rearm, רצפות-עומק).
   678→  // חילוף-הכיוונים שומר עליה בת-קיימא: מחזור-זוגי היא קונה SAOS, אי-זוגי
   679→  // מוכרת — נטו-שטוח, המלאי ממוחזר בינה לבין סוכן-ההתעמלות ולא מתרוקן.
   680→  if (tick % EXERCISE_PAIR_TICK === 12) {
   681→    const cycle = Math.floor(tick / EXERCISE_PAIR_TICK);
   682→    const botSells = cycle % 2 === 0;
   683→    const seller = botSells ? "bot-e" : "arena";
   684→    const buyer = botSells ? "arena" : "bot-e";
   685→    const sellPx = Math.floor((fair * 9_970) / 10_000); // −0.3% מתחת לפייר
   686→    const buyPx = Math.floor((fair * 10_030) / 10_000);  // +0.3% מעליו — חוצה
   687→    const sellOk = avail(st, seller, m.base) >= EXERCISE_COW_AMT;
   688→    const buyOk = avail(st, buyer, m.quote) >= notional(buyPx, EXERCISE_COW_AMT);
   689→    if (sellPx > 0 && buyPx >= sellPx && sellOk && buyOk) {
   690→      submit(rt, st, seller, { type: "cow_order", base: m.base, quote: m.quote, side: "sell", price: sellPx, amount: EXERCISE_COW_AMT }, pushOp);
   691→      submit(rt, st, buyer, { type: "cow_order", base: m.base, quote: m.quote, side: "buy", price: buyPx, amount: EXERCISE_COW_AMT }, pushOp);
   692→    }
   693→  }
   694→
   695→  // ── DCA: תוכנית קטנה, רק אם אין אחת פעילה לסוכן ──
   696→  if (tick % EXERCISE_PLAN_TICK === 60) {
   697→    const mineActive = Object.values(st.dca ?? {}).some((p) => p.owner === "bot-e" && p.remainingMu > 0);
   698→    const cost = EXERCISE_DCA_PER_SLICE * EXERCISE_DCA_SLICES;
   699→    if (!mineActive && avail(st, "bot-e", "USDS") >= cost) {
   700→      submit(rt, st, "bot-e", { type: "dca_create", asset: "USDS", to: "SAOS", perSlice: EXERCISE_DCA_PER_SLICE, slicesMax: EXERCISE_DCA_SLICES, everyK: 40, minOutBps: 100 }, pushOp);
   701→    }
   702→  }
   703→
   704→  // ── טריגר: כיוון מתחלף, 30bps נגד הפייר, תוקף 500 בלוקים ──
   705→  if (tick % EXERCISE_PLAN_TICK === 300) {
   706→    const planIdx = Math.floor(tick / EXERCISE_PLAN_TICK);
   707→    const side: "sell" | "buy" = planIdx % 2 === 0 ? "sell" : "buy";
   708→    const triggerPx = side === "sell"
   709→      ? Math.floor((fair * 9_970) / 10_000)   // נורה אם המחיר יורד 0.3%
   710→      : Math.floor((fair * 10_030) / 10_000);  // נורה אם המחיר עולה 0.3%
   711→    if (triggerPx > 0) {
   712→      if (side === "sell" && avail(st, "bot-e", m.base) >= EXERCISE_TRIG_AMT) {
   713→        submit(rt, st, "bot-e", { type: "trig_order", base: m.base, quote: m.quote, side, triggerPx, limitPx: 0, amount: EXERCISE_TRIG_AMT, expiresH: st.height + 500 }, pushOp);
   714→      } else if (side === "buy" && avail(st, "bot-e", m.quote) >= notional(triggerPx, EXERCISE_TRIG_AMT)) {
   715→        submit(rt, st, "bot-e", { type: "trig_order", base: m.base, quote: m.quote, side, triggerPx, limitPx: 0, amount: EXERCISE_TRIG_AMT, expiresH: st.height + 500 }, pushOp);
   716→      }
   717→    }
   718→  }
